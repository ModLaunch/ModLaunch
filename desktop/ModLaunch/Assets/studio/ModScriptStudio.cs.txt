// ModScript Studio — конструктор внутри игры (Unity + BepInEx 5). Собирается ModLaunch под каждую игру
// (её собственные UnityEngine и BepInEx), поэтому одна и та же версия подходит играм на Unity 2017–6.
// Режим разработчика (файл dev.flag рядом с настройками) — кнопка «Создать мод» в игре и конструктор;
// без него плагин только загружает сохранённые карты: так сделанное становится обычным модом.
// Нарочно без UnityEngine.Input (игры на новом Input System его выключают), без встроенного шрифта IMGUI
// (в Unity 6 его в игре может не быть — берём шрифт Windows) и без API новее .NET 3.5.
// Три части одного класса: эта — карта и её сохранение; Studio.Tools — выбор, стрелки, правки, свойства;
// Studio.Ui — окно редактора (лента, проводник, свойства, библиотека, вывод), камера и самопроверка.
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;
using BepInEx;
using UnityEngine;
using UnityEngine.SceneManagement;

/// <summary>
/// Поведение объекта, как маленький скрипт в Roblox: вращение (по любой оси), парение, пульс, маятник, качание, орбита,
/// блуждание, мигание, радуга, мерцание света, поворот к камере. Сохраняется вместе с картой строкой Code().
/// </summary>
public class ModScriptBehaviour : MonoBehaviour
{
    public float Spin, Bob, BobSpeed = 1.5f, Pulse, Swing, Orbit, Blink, Rainbow;
    public int SpinAxis = 1;
    public bool Flicker, Wobble, Billboard, Drift, Paused;
    Vector3 _base, _scale;
    Quaternion _rot;
    float _t;
    Renderer[] _rs;
    Light[] _ls;
    float[] _li;
    Material[] _mats;

    static readonly CultureInfo Inv = CultureInfo.InvariantCulture;

    public void Rebase()
    {
        _base = transform.localPosition;
        _scale = transform.localScale;
        _rot = transform.localRotation;
        _t = 0;
        _rs = GetComponentsInChildren<Renderer>();
        _ls = GetComponentsInChildren<Light>();
        _li = _ls.Select(l => l.intensity).ToArray();
        _mats = null;
    }

    /// <summary>Остановить и вернуть исходное (редактор правит исходное положение, а не середину движения).</summary>
    public void Pause()
    {
        if (Paused) return;
        Paused = true;
        if (_rs == null) return;
        transform.localPosition = _base;
        transform.localScale = _scale;
        transform.localRotation = _rot;
        foreach (var r in _rs) if (r != null) r.enabled = true;
        for (var i = 0; i < _ls.Length; i++) if (_ls[i] != null) { _ls[i].enabled = true; _ls[i].intensity = _li[i]; }
    }

    public void Resume() { Rebase(); Paused = false; }

    void Start() { if (_rs == null) Rebase(); }

    static Vector3 Ax(int i) { return i == 0 ? Vector3.right : i == 2 ? Vector3.forward : Vector3.up; }

    void Update()
    {
        if (Paused || _rs == null) return;
        var dt = Time.deltaTime;
        _t += dt;
        if (Spin != 0) transform.Rotate(Ax(SpinAxis) * Spin * dt, Space.World);
        if (Bob != 0 || Orbit != 0 || Drift)
        {
            var p = _base;
            if (Bob != 0) p += Vector3.up * Mathf.Sin(_t * BobSpeed * Mathf.PI) * Bob;
            if (Orbit != 0) p += new Vector3(Mathf.Cos(_t * 0.8f), 0, Mathf.Sin(_t * 0.8f)) * Orbit;
            if (Drift) p += new Vector3(Mathf.PerlinNoise(_t * 0.25f, 1.7f) - 0.5f, 0, Mathf.PerlinNoise(3.1f, _t * 0.25f) - 0.5f) * 6f;
            transform.localPosition = p;
        }
        if (Pulse != 0) transform.localScale = _scale * (1 + Mathf.Sin(_t * 3f) * Pulse);
        if ((Swing != 0 || Wobble) && Spin == 0)
        {
            var q = _rot;
            if (Swing != 0) q = q * Quaternion.Euler(0, 0, Mathf.Sin(_t * 2f) * Swing);
            if (Wobble) q = q * Quaternion.Euler(Mathf.Sin(_t * 3f) * 7f, 0, Mathf.Cos(_t * 2.3f) * 7f);
            transform.localRotation = q;
        }
        if (Blink > 0)
        {
            var on = Mathf.Repeat(_t, Blink) < Blink * 0.5f;
            foreach (var r in _rs) if (r != null) r.enabled = on;
            foreach (var l in _ls) if (l != null) l.enabled = on;
        }
        if (Rainbow > 0)
        {
            var c = Color.HSVToRGB(Mathf.Repeat(_t * Rainbow * 0.1f, 1f), 0.75f, 1f);
            if (_mats == null) _mats = _rs.Where(r => r != null && !(r is ParticleSystemRenderer)).SelectMany(r => r.materials).ToArray();
            foreach (var m in _mats) { if (m.HasProperty("_BaseColor")) m.SetColor("_BaseColor", c); if (m.HasProperty("_Color")) m.color = c; }
            foreach (var l in _ls) if (l != null) l.color = c;
        }
        if (Flicker)
            for (var i = 0; i < _ls.Length; i++) if (_ls[i] != null) _ls[i].intensity = _li[i] * (0.65f + 0.7f * Mathf.PerlinNoise(_t * 9f, i));
        if (Billboard)
        {
            var cam = Camera.main;
            if (cam != null)
            {
                var d = cam.transform.position - transform.position;
                d.y = 0;
                if (d.sqrMagnitude > 0.01f) transform.rotation = Quaternion.LookRotation(-d);
            }
        }
    }

    static string N(float v) { return v.ToString("R", Inv); }

    public bool Empty { get { return Code() == ""; } }

    public string Code()
    {
        var p = new List<string>();
        if (Spin != 0) p.Add("spin=" + N(Spin) + "," + SpinAxis);
        if (Bob != 0) p.Add("bob=" + N(Bob) + "," + N(BobSpeed));
        if (Pulse != 0) p.Add("pulse=" + N(Pulse));
        if (Swing != 0) p.Add("swing=" + N(Swing));
        if (Orbit != 0) p.Add("orbit=" + N(Orbit));
        if (Blink != 0) p.Add("blink=" + N(Blink));
        if (Rainbow != 0) p.Add("rainbow=" + N(Rainbow));
        if (Flicker) p.Add("flicker");
        if (Wobble) p.Add("wobble");
        if (Billboard) p.Add("billboard");
        if (Drift) p.Add("drift");
        return string.Join(";", p.ToArray());
    }

    public void Parse(string code)
    {
        Spin = Bob = Pulse = Swing = Orbit = Blink = Rainbow = 0;
        SpinAxis = 1;
        Flicker = Wobble = Billboard = Drift = false;
        foreach (var part in (code ?? "").Split(';'))
        {
            var kv = part.Split('=');
            var key = kv[0].Trim();
            if (kv.Length == 1)
            {
                if (key == "flicker") Flicker = true;
                if (key == "wobble") Wobble = true;
                if (key == "billboard") Billboard = true;
                if (key == "drift") Drift = true;
                continue;
            }
            var v = kv[1].Split(',');
            float a;
            if (!float.TryParse(v[0], NumberStyles.Float, Inv, out a)) continue;
            float b = 0;
            var hasB = v.Length > 1 && float.TryParse(v[1], NumberStyles.Float, Inv, out b);
            switch (key)
            {
                case "spin": Spin = a; if (hasB) SpinAxis = (int)b; break;
                case "bob": Bob = a; if (hasB) BobSpeed = b; break;
                case "pulse": Pulse = a; break;
                case "swing": Swing = a; break;
                case "orbit": Orbit = a; break;
                case "blink": Blink = a; break;
                case "rainbow": Rainbow = a; break;
            }
        }
    }
}

[BepInPlugin("modlaunch.studio", "ModScript Studio", "23.8.0")]
public partial class ModScriptStudio : BaseUnityPlugin
{
    // ------------------------------------------------------------------ файлы и строки

    static readonly CultureInfo Inv = CultureInfo.InvariantCulture;
    string _root;
    bool _dev;
    bool _autotest;
    readonly Dictionary<string, string> _l = new Dictionary<string, string>();

    string T(string key, string fallback)
    {
        string v;
        return _l.TryGetValue(key, out v) ? v : fallback;
    }

    void Awake()
    {
        _root = Path.Combine(Paths.ConfigPath, "ModScriptStudio");
        try { Directory.CreateDirectory(Path.Combine(_root, "textures")); Directory.CreateDirectory(Path.Combine(_root, "models")); } catch { }
        _dev = File.Exists(Path.Combine(_root, "dev.flag"));
        var auto = Path.Combine(_root, "autotest.flag");
        if (File.Exists(auto)) { _autotest = true; try { File.Delete(auto); } catch { } HoldQuit(); }
        try
        {
            var labels = Path.Combine(_root, "labels.txt");
            if (File.Exists(labels))
                foreach (var line in File.ReadAllLines(labels, Encoding.UTF8))
                {
                    var eq = line.IndexOf('=');
                    if (eq > 0) _l[line.Substring(0, eq).Trim()] = line.Substring(eq + 1).Replace("\\n", "\n");
                }
        }
        catch { }
        SceneManager.sceneLoaded += OnSceneLoaded;
        if (_dev) { Application.logMessageReceived += OnLog; LoadProfile(); LoadPrefs(); }
        Logger.LogInfo("ModScript Studio ready, developer mode: " + _dev + (_autotest ? ", self-test" : "") + ", Unity " + Application.unityVersion);
    }

    void OnDestroy()
    {
        SceneManager.sceneLoaded -= OnSceneLoaded;
        Application.logMessageReceived -= OnLog;
    }

    /// <summary>Уровень, загруженный раньше плагина, тоже получает свою сохранённую карту.</summary>
    void Start()
    {
        var active = SceneManager.GetActiveScene();
        if (_scene == "" && active.isLoaded) { _scene = active.name; Load(); }
    }

    // ------------------------------------------------------------------ карта (то, что сделано в конструкторе)

    /// <summary>
    /// Поставленный объект: откуда (prim:Cube, asset:имя, clone:путь, obj:файл), где, материал, физика, поведение.
    /// Parent — номер группы (тоже поставленной): тогда положение, поворот и размер — относительно неё.
    /// </summary>
    class Placed
    {
        public string Id, Source, Name, Mat = "", Beh = "", Parent = "", Look = "", PhysMat = "";
        /// <summary>Исходные материалы частей: от них заново накладываются материал, цвет и вид (и отменяются).</summary>
        public Renderer[] OrigR;
        public Material[][] OrigM;
        public Vector3 Pos, Scale = Vector3.one;
        public Quaternion Rot = Quaternion.identity;
        public Color Tint = Color.white;
        public bool Tinted, Physics;
        public GameObject Go;
    }

    /// <summary>Изменённый объект уровня: скрыт или сдвинут.</summary>
    class Touched
    {
        public string Path;
        public bool Hidden, Moved;
        public Vector3 Pos, Scale;
        public Quaternion Rot;
        public GameObject Go;
    }

    /// <summary>Изменённое свойство компонента (как в окне Properties): кому (id:номер или путь), компонент, поле, значение.</summary>
    class Prop
    {
        public string Target, Component, Member, Value;
    }

    /// <summary>Настройки мира уровня: солнце, окружающий свет, туман (-1 — не меняли).</summary>
    class World
    {
        public float SunIntensity = -1, SunPitch = -999, SunYaw = -999, Ambient = -1, FogDensity = -1;
        public int Fog = -1;
        public Color SunColor = Unset, AmbientColor = Unset, FogColor = Unset;
        public float Shadow = -1, Cycle = -1;
        public string Weather = "";
        public bool Changed { get { return SunIntensity >= 0 || SunPitch > -999 || Ambient >= 0 || Fog >= 0 || FogDensity >= 0 || Changed2; } }
        public bool Changed2 { get { return SunColor.a >= 0 || AmbientColor.a >= 0 || FogColor.a >= 0 || Shadow >= 0; } }
        public bool Changed3 { get { return Weather != "" || Cycle > 0; } }
        public static readonly Color Unset = new Color(0, 0, 0, -1);
    }

    readonly List<Placed> _placed = new List<Placed>();
    readonly List<Touched> _touched = new List<Touched>();
    readonly List<Prop> _props = new List<Prop>();
    World _world = new World();
    string _scene = "";
    bool _dirty;
    float _dirtyAt;

    /// <summary>Сохранить чуть позже (правки идут подряд — пишем файл один раз).</summary>
    void Dirty() { _dirty = true; _dirtyAt = Time.realtimeSinceStartup; }

    string LayoutFile(string scene)
    {
        var safe = new StringBuilder();
        foreach (var c in scene) safe.Append(char.IsLetterOrDigit(c) || c == '-' || c == '_' ? c : '_');
        return Path.Combine(_root, (safe.Length == 0 ? "scene" : safe.ToString()) + ".layout");
    }

    static string V(Vector3 v) { return v.x.ToString("R", Inv) + "," + v.y.ToString("R", Inv) + "," + v.z.ToString("R", Inv); }
    static string Q(Quaternion q) { return q.x.ToString("R", Inv) + "," + q.y.ToString("R", Inv) + "," + q.z.ToString("R", Inv) + "," + q.w.ToString("R", Inv); }
    static string C(Color c) { return c.r.ToString("R", Inv) + "," + c.g.ToString("R", Inv) + "," + c.b.ToString("R", Inv) + "," + c.a.ToString("R", Inv); }
    static float[] F(string s) { return s.Split(',').Select(x => float.Parse(x, Inv)).ToArray(); }
    static string Esc(string s) { return (s ?? "").Replace("|", "/").Replace("\n", " "); }

    /// <summary>Положение для файла: у объекта в группе — относительно группы.</summary>
    void Capture(Placed p)
    {
        if (p.Go == null) return;
        var b = p.Go.GetComponent<ModScriptBehaviour>();
        p.Beh = b != null ? b.Code() : "";
        var parent = p.Go.transform.parent != null ? PlacedExact(p.Go.transform.parent.gameObject) : null;
        p.Parent = parent != null ? parent.Id : "";
        if (parent != null) { p.Pos = p.Go.transform.localPosition; p.Rot = p.Go.transform.localRotation; }
        else { p.Pos = p.Go.transform.position; p.Rot = p.Go.transform.rotation; }
        p.Scale = p.Go.transform.localScale;
        p.Name = p.Go.name;
    }

    void Save()
    {
        _dirty = false;
        // Самопроверка ничего не сохраняет: её кубик не должен попасть в карту игрока.
        if (_scene == "" || _autotest) return;
        try
        {
            var lines = new List<string> { "# ModScript Studio layout v4 — scene " + _scene };
            foreach (var p in _placed)
            {
                Capture(p);
                lines.Add("place|" + p.Id + "|" + Esc(p.Source) + "|" + Esc(p.Name) + "|" + V(p.Pos) + "|" + Q(p.Rot) + "|" + V(p.Scale) + "|" + Esc(p.Mat) + "|"
                    + (p.Tinted ? C(p.Tint) : "") + "|" + (p.Physics ? "phys" : "") + "|" + Esc(p.Beh) + "|" + p.Parent + "|" + Esc(p.Look) + "|" + Esc(p.PhysMat));
            }
            foreach (var t in _touched)
            {
                if (t.Go != null && t.Moved) { t.Pos = t.Go.transform.position; t.Rot = t.Go.transform.rotation; t.Scale = t.Go.transform.localScale; }
                if (t.Hidden) lines.Add("hide|" + Esc(t.Path));
                else if (t.Moved) lines.Add("move|" + Esc(t.Path) + "|" + V(t.Pos) + "|" + Q(t.Rot) + "|" + V(t.Scale));
            }
            foreach (var pr in _props)
                lines.Add("prop|" + Esc(pr.Target) + "|" + Esc(pr.Component) + "|" + Esc(pr.Member) + "|" + Esc(pr.Value));
            if (_world.Changed)
                lines.Add("world|" + _world.SunIntensity.ToString("R", Inv) + "|" + _world.SunPitch.ToString("R", Inv) + "|" + _world.SunYaw.ToString("R", Inv) + "|" + _world.Ambient.ToString("R", Inv) + "|" + _world.Fog + "|" + _world.FogDensity.ToString("R", Inv));
            if (_world.Changed2)
                lines.Add("world2|" + C(_world.SunColor) + "|" + C(_world.AmbientColor) + "|" + C(_world.FogColor) + "|" + _world.Shadow.ToString("R", Inv));
            if (_world.Changed3) lines.Add("weather|" + Esc(_world.Weather) + "|" + _world.Cycle.ToString("R", Inv));
            foreach (var kv in _bookmarks)
                lines.Add("cam|" + kv.Key + "|" + V(kv.Value.Key) + "|" + Q(kv.Value.Value));
            File.WriteAllLines(LayoutFile(_scene), lines.ToArray(), Encoding.UTF8);
            Status(T("saved", "Saved"));
        }
        catch (Exception e) { Status(e.Message); Logger.LogWarning("save: " + e); }
    }

    void OnSceneLoaded(Scene scene, LoadSceneMode mode)
    {
        if (_autotest && _testStage < TestLast)
        {
            // Новая загрузка посреди проверки — начать заново, когда уровень успокоится.
            _testStage = 0;
            _testAt = Time.realtimeSinceStartup + 10f;
        }
        if (mode != LoadSceneMode.Single && _scene != "") return;
        if (_open) Close();
        _scene = scene.name;
        _placed.Clear();
        _touched.Clear();
        _props.Clear();
        _world = new World();
        _bookmarks.Clear();
        _library = null;
        _materials = null;
        _selection.Clear();
        _undo.Clear();
        _redo.Clear();
        Load();
    }

    /// <summary>Поставить сохранённое: объекты (и группы), скрытое и сдвинутое, свойства компонентов, мир.</summary>
    void Load()
    {
        var file = LayoutFile(_scene);
        if (!File.Exists(file)) return;
        string[] lines;
        try { lines = File.ReadAllLines(file, Encoding.UTF8); } catch { return; }
        foreach (var line in lines)
        {
            if (line.StartsWith("#") || line.Trim() == "") continue;
            var f = line.Split('|');
            try
            {
                if (f[0] == "place" && f.Length >= 8)
                {
                    var p = new Placed { Id = f[1], Source = f[2], Name = f[3], Mat = f[7] };
                    var pos = F(f[4]); var rot = F(f[5]); var sc = F(f[6]);
                    p.Pos = new Vector3(pos[0], pos[1], pos[2]);
                    p.Rot = new Quaternion(rot[0], rot[1], rot[2], rot[3]);
                    p.Scale = new Vector3(sc[0], sc[1], sc[2]);
                    if (f.Length > 8 && f[8] != "") { var c = F(f[8]); p.Tint = new Color(c[0], c[1], c[2], c[3]); p.Tinted = true; }
                    p.Physics = f.Length > 9 && f[9] == "phys";
                    p.Beh = f.Length > 10 ? f[10] : "";
                    p.Parent = f.Length > 11 ? f[11] : "";
                    p.Look = f.Length > 12 ? f[12] : "";
                    p.PhysMat = f.Length > 13 ? f[13] : "";
                    p.Go = Make(p.Source, p.Pos, p.Rot);
                    if (p.Go == null) continue;
                    Dress(p);
                    _placed.Add(p);
                }
                else if (f[0] == "hide" && f.Length >= 2)
                {
                    var go = FindPath(f[1]);
                    if (go != null) go.SetActive(false);
                    _touched.Add(new Touched { Path = f[1], Hidden = true, Go = go });
                }
                else if (f[0] == "move" && f.Length >= 5)
                {
                    var go = FindPath(f[1]);
                    var pos = F(f[2]); var rot = F(f[3]); var sc = F(f[4]);
                    var t = new Touched { Path = f[1], Moved = true, Go = go, Pos = new Vector3(pos[0], pos[1], pos[2]), Rot = new Quaternion(rot[0], rot[1], rot[2], rot[3]), Scale = new Vector3(sc[0], sc[1], sc[2]) };
                    if (go != null) { go.transform.position = t.Pos; go.transform.rotation = t.Rot; go.transform.localScale = t.Scale; }
                    _touched.Add(t);
                }
                else if (f[0] == "prop" && f.Length >= 5)
                    _props.Add(new Prop { Target = f[1], Component = f[2], Member = f[3], Value = f[4] });
                else if (f[0] == "world" && f.Length >= 7)
                {
                    _world = new World
                    {
                        SunIntensity = float.Parse(f[1], Inv), SunPitch = float.Parse(f[2], Inv), SunYaw = float.Parse(f[3], Inv),
                        Ambient = float.Parse(f[4], Inv), Fog = int.Parse(f[5], Inv), FogDensity = float.Parse(f[6], Inv),
                    };
                    ApplyWorld();
                }
                else if (f[0] == "world2" && f.Length >= 5)
                {
                    var sc = F(f[1]); var ac = F(f[2]); var fc = F(f[3]);
                    _world.SunColor = new Color(sc[0], sc[1], sc[2], sc[3]);
                    _world.AmbientColor = new Color(ac[0], ac[1], ac[2], ac[3]);
                    _world.FogColor = new Color(fc[0], fc[1], fc[2], fc[3]);
                    _world.Shadow = float.Parse(f[4], Inv);
                    ApplyWorld();
                }
                else if (f[0] == "weather" && f.Length >= 3)
                {
                    _world.Weather = f[1];
                    _world.Cycle = float.Parse(f[2], Inv);
                    ApplyWorld();
                }
                else if (f[0] == "cam" && f.Length >= 4)
                {
                    var pos = F(f[2]); var rot = F(f[3]);
                    _bookmarks[int.Parse(f[1], Inv)] = new KeyValuePair<Vector3, Quaternion>(new Vector3(pos[0], pos[1], pos[2]), new Quaternion(rot[0], rot[1], rot[2], rot[3]));
                }
            }
            catch (Exception e) { Logger.LogWarning("layout line skipped: " + line + " — " + e.Message); }
        }
        // Группы: объекты встают в свою группу, положение — относительно неё.
        foreach (var p in _placed)
        {
            if (p.Parent == "") continue;
            var parent = _placed.FirstOrDefault(x => x.Id == p.Parent);
            if (parent == null || parent.Go == null) continue;
            p.Go.transform.SetParent(parent.Go.transform, false);
            p.Go.transform.localPosition = p.Pos;
            p.Go.transform.localRotation = p.Rot;
            p.Go.transform.localScale = p.Scale;
            var b = p.Go.GetComponent<ModScriptBehaviour>();
            if (b != null) b.Rebase();
        }
        foreach (var pr in _props) ApplyProp(pr);
        Logger.LogInfo("ModScript Studio: " + _placed.Count + " objects, " + _props.Count + " properties in " + _scene);
    }

    /// <summary>Имя, размер, материал, цвет, физика и поведение поставленного объекта.</summary>
    void Dress(Placed p)
    {
        p.Go.name = p.Name;
        p.Go.transform.localScale = p.Scale;
        CaptureOrig(p);
        ApplyStyle(p);
        SetPhysics(p.Go, p.Physics);
        ApplyPhysMat(p.Go, p.PhysMat);
        if (p.Beh != "")
        {
            var b = p.Go.GetComponent<ModScriptBehaviour>();
            if (b == null) b = p.Go.AddComponent<ModScriptBehaviour>();
            b.Parse(p.Beh);
            b.Rebase();
        }
    }

    readonly Dictionary<int, KeyValuePair<Vector3, Quaternion>> _bookmarks = new Dictionary<int, KeyValuePair<Vector3, Quaternion>>();

    /// <summary>Запомнить исходные материалы частей объекта (один раз).</summary>
    static void CaptureOrig(Placed p)
    {
        if (p.OrigR != null || p.Go == null) return;
        p.OrigR = p.Go.GetComponentsInChildren<Renderer>(true).Where(r => !(r is ParticleSystemRenderer)).ToArray();
        p.OrigM = p.OrigR.Select(r => r.sharedMaterials).ToArray();
    }

    /// <summary>Наложить заново: исходные материалы → материал → цвет → вид (металл, глянец, свечение).</summary>
    void ApplyStyle(Placed p)
    {
        if (p.Go == null) return;
        CaptureOrig(p);
        for (var i = 0; i < p.OrigR.Length; i++) if (p.OrigR[i] != null) p.OrigR[i].sharedMaterials = p.OrigM[i];
        if (p.Mat != "") ApplyMaterial(p.Go, p.Mat);
        if (p.Tinted) ApplyTint(p.Go, p.Tint);
        else foreach (var l in p.Go.GetComponentsInChildren<Light>(true)) if (p.Source.StartsWith("prim:")) l.color = Color.white;
        if (p.Look != "") ApplyLook(p.Go, p.Look);
    }

    /// <summary>Вид материала: m — металличность, s — гладкость, e — свечение (цвет). Строка вида «m=1;s=0.8;e=1,0.6,0.1».</summary>
    static void ApplyLook(GameObject go, string look)
    {
        float metal = -1, smooth = -1;
        Color glow = new Color(0, 0, 0, -1);
        foreach (var part in look.Split(';'))
        {
            var kv = part.Split('=');
            if (kv.Length != 2) continue;
            if (kv[0] == "m") metal = float.Parse(kv[1], Inv);
            if (kv[0] == "s") smooth = float.Parse(kv[1], Inv);
            if (kv[0] == "e") { var c = F(kv[1]); glow = new Color(c[0], c[1], c[2], 1); }
        }
        foreach (var r in go.GetComponentsInChildren<Renderer>(true))
        {
            if (r is ParticleSystemRenderer) continue;
            try
            {
                foreach (var m in r.materials)
                {
                    if (metal >= 0 && m.HasProperty("_Metallic")) m.SetFloat("_Metallic", metal);
                    if (smooth >= 0) { if (m.HasProperty("_Smoothness")) m.SetFloat("_Smoothness", smooth); if (m.HasProperty("_Glossiness")) m.SetFloat("_Glossiness", smooth); }
                    if (glow.a >= 0 && m.HasProperty("_EmissionColor"))
                    {
                        m.EnableKeyword("_EMISSION");
                        m.globalIlluminationFlags = MaterialGlobalIlluminationFlags.RealtimeEmissive;
                        m.SetColor("_EmissionColor", glow * 2.2f);
                    }
                }
            }
            catch { }
        }
    }

    /// <summary>Упругость и трение коллайдеров (PhysicMaterial в старых Unity, PhysicsMaterial в Unity 6 — ищем отражением).</summary>
    static void ApplyPhysMat(GameObject go, string spec)
    {
        if (string.IsNullOrEmpty(spec)) return;
        var type = Type.GetType("UnityEngine.PhysicsMaterial, UnityEngine.PhysicsModule") ?? Type.GetType("UnityEngine.PhysicMaterial, UnityEngine.PhysicsModule") ?? Type.GetType("UnityEngine.PhysicMaterial, UnityEngine");
        if (type == null) return;
        try
        {
            var mat = Activator.CreateInstance(type);
            foreach (var part in spec.Split(';'))
            {
                var kv = part.Split('=');
                if (kv.Length != 2) continue;
                var v = float.Parse(kv[1], Inv);
                if (kv[0] == "b") { type.GetProperty("bounciness").SetValue(mat, v, null); SetEnumProp(type, mat, "bounceCombine", v > 0.5f ? "Maximum" : "Average"); }
                if (kv[0] == "f") { type.GetProperty("dynamicFriction").SetValue(mat, v, null); type.GetProperty("staticFriction").SetValue(mat, v, null); SetEnumProp(type, mat, "frictionCombine", v < 0.1f ? "Minimum" : "Average"); }
            }
            foreach (var c in go.GetComponentsInChildren<Collider>(true))
            {
                var prop = c.GetType().GetProperty("sharedMaterial");
                if (prop != null) prop.SetValue(c, mat, null);
            }
        }
        catch { }
    }

    static void SetEnumProp(Type type, object o, string name, string value)
    {
        var p = type.GetProperty(name);
        if (p != null && p.PropertyType.IsEnum) p.SetValue(o, Enum.Parse(p.PropertyType, value), null);
    }

    /// <summary>Создать объект по источнику: фигура, ассет игры (по имени), копия объекта уровня (по пути), модель OBJ.</summary>
    GameObject Make(string source, Vector3 pos, Quaternion rot)
    {
        GameObject go = null;
        if (source.StartsWith("prim:"))
        {
            var kind = source.Substring(5);
            if (kind == "Light" || kind == "Spot")
            {
                go = new GameObject(kind == "Spot" ? "Spot light" : "Light");
                var l = go.AddComponent<Light>();
                l.type = kind == "Spot" ? LightType.Spot : LightType.Point;
                l.range = 15;
                l.intensity = 2;
                if (kind == "Spot") { l.spotAngle = 50; go.transform.rotation = Quaternion.Euler(90, 0, 0); }
            }
            else if (kind == "Empty") go = new GameObject("Group");
            else
            {
                go = GameObject.CreatePrimitive((PrimitiveType)Enum.Parse(typeof(PrimitiveType), kind));
                // У фигуры — материал игры, а не розовый «нет шейдера» (в играх на URP/HDRP стандартный шейдер не работает).
                var basis = BaseMaterial();
                if (basis != null) go.GetComponent<Renderer>().sharedMaterial = basis;
            }
        }
        else if (source.StartsWith("asset:"))
        {
            var name = source.Substring(6);
            var asset = Resources.FindObjectsOfTypeAll<GameObject>().FirstOrDefault(g => g.name == name && !g.scene.IsValid() && g.transform.parent == null);
            if (asset != null) go = (GameObject)Instantiate(asset);
        }
        else if (source.StartsWith("clone:"))
        {
            var orig = FindPath(source.Substring(6));
            if (orig != null) go = (GameObject)Instantiate(orig);
        }
        else if (source.StartsWith("obj:")) go = LoadObj(source.Substring(4));
        else if (source.StartsWith("kit:")) go = Kit(source.Substring(4));
        if (go == null) return null;
        go.SetActive(true);
        go.transform.position = pos;
        if (!source.StartsWith("prim:Spot")) go.transform.rotation = rot;
        return go;
    }

    Material _base;

    /// <summary>Непрозрачный материал самой игры с обычным освещением — основа для фигур, моделей и своих текстур.</summary>
    Material BaseMaterial()
    {
        if (_base != null) return _base;
        string[] wanted = { "Universal Render Pipeline/Lit", "HDRP/Lit", "Standard", "Universal Render Pipeline/Simple Lit", "Legacy Shaders/Diffuse", "Sprites/Default" };
        foreach (var name in wanted)
        {
            var sh = Shader.Find(name);
            if (sh == null) continue;
            _base = new Material(sh) { name = "ModScript Studio" };
            return _base;
        }
        foreach (var r in FindObjectsOfType<MeshRenderer>())
            if (r.sharedMaterial != null && r.sharedMaterial.shader != null) { _base = new Material(r.sharedMaterial) { name = "ModScript Studio" }; return _base; }
        return null;
    }

    static void SetPhysics(GameObject go, bool on)
    {
        foreach (var rb in go.GetComponentsInChildren<Rigidbody>(true))
        {
            // Скорость обнуляем, пока тело ещё не кинематическое (у кинематического Unity 6 ругается).
            if (!on && !rb.isKinematic) { rb.velocity = Vector3.zero; rb.angularVelocity = Vector3.zero; }
            rb.isKinematic = !on;
        }
        if (on && go.GetComponentInChildren<Rigidbody>(true) == null && go.GetComponentInChildren<Collider>(true) != null) go.AddComponent<Rigidbody>();
    }

    static string PathOf(Transform t)
    {
        var parts = new List<string>();
        while (t != null)
        {
            var name = t.name;
            if (t.parent != null)
            {
                var same = 0;
                for (var i = 0; i < t.GetSiblingIndex(); i++) if (t.parent.GetChild(i).name == name) same++;
                if (same > 0) name += "#" + same;
            }
            parts.Insert(0, name);
            t = t.parent;
        }
        return string.Join("/", parts.ToArray());
    }

    static GameObject FindPath(string path)
    {
        var parts = path.Split('/');
        for (var s = 0; s < SceneManager.sceneCount; s++)
        {
            var scene = SceneManager.GetSceneAt(s);
            if (!scene.isLoaded) continue;
            foreach (var root in scene.GetRootGameObjects())
            {
                if (root.name != parts[0]) continue;
                var t = root.transform;
                for (var i = 1; i < parts.Length && t != null; i++)
                {
                    var name = parts[i];
                    var index = 0;
                    var hash = name.LastIndexOf('#');
                    if (hash > 0) { int.TryParse(name.Substring(hash + 1), out index); name = name.Substring(0, hash); }
                    Transform hit = null;
                    for (var c = 0; c < t.childCount; c++)
                        if (t.GetChild(c).name == name) { if (index == 0) { hit = t.GetChild(c); break; } index--; }
                    t = hit;
                }
                if (t != null) return t.gameObject;
            }
        }
        return null;
    }

    // ------------------------------------------------------------------ свои модели (OBJ) и текстуры

    /// <summary>
    /// Модель OBJ из папки models: вершины, развёртка, нормали, многоугольники (режутся на треугольники).
    /// Текстура — картинка с тем же именем (model.png / model.jpg) рядом. Большие модели — по 65 тыс. вершин на часть.
    /// </summary>
    GameObject LoadObj(string file)
    {
        var full = Path.Combine(Path.Combine(_root, "models"), Path.GetFileName(file));
        if (!File.Exists(full)) return null;
        var v = new List<Vector3>(); var vt = new List<Vector2>(); var vn = new List<Vector3>();
        var outV = new List<Vector3>(); var outT = new List<Vector2>(); var outN = new List<Vector3>();
        foreach (var raw in File.ReadAllLines(full))
        {
            var line = raw.Trim();
            if (line.Length < 2 || line[0] == '#') continue;
            var p = line.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
            if (p[0] == "v" && p.Length >= 4) v.Add(new Vector3(-float.Parse(p[1], Inv), float.Parse(p[2], Inv), float.Parse(p[3], Inv)));
            else if (p[0] == "vt" && p.Length >= 3) vt.Add(new Vector2(float.Parse(p[1], Inv), float.Parse(p[2], Inv)));
            else if (p[0] == "vn" && p.Length >= 4) vn.Add(new Vector3(-float.Parse(p[1], Inv), float.Parse(p[2], Inv), float.Parse(p[3], Inv)));
            else if (p[0] == "f" && p.Length >= 4)
            {
                for (var k = 2; k + 1 < p.Length; k++)
                    foreach (var corner in new[] { p[1], p[k + 1], p[k] })
                    {
                        var idx = corner.Split('/');
                        var vi = Index(idx[0], v.Count);
                        outV.Add(v[vi]);
                        outT.Add(idx.Length > 1 && idx[1] != "" ? vt[Index(idx[1], vt.Count)] : Vector2.zero);
                        outN.Add(idx.Length > 2 && idx[2] != "" ? vn[Index(idx[2], vn.Count)] : Vector3.zero);
                    }
            }
        }
        if (outV.Count == 0) return null;
        var root = new GameObject(Path.GetFileNameWithoutExtension(file));
        var mat = BaseMaterial() != null ? new Material(BaseMaterial()) : null;
        foreach (var ext in new[] { ".png", ".jpg", ".jpeg" })
        {
            var texFile = Path.Combine(Path.Combine(_root, "models"), Path.GetFileNameWithoutExtension(file) + ext);
            if (mat == null || !File.Exists(texFile)) continue;
            var tex = new Texture2D(2, 2);
            if (LoadImage(tex, File.ReadAllBytes(texFile))) { mat.mainTexture = tex; if (mat.HasProperty("_BaseMap")) mat.SetTexture("_BaseMap", tex); }
            break;
        }
        const int chunk = 64998;
        for (var start = 0; start < outV.Count; start += chunk)
        {
            var count = Math.Min(chunk, outV.Count - start);
            var part = new GameObject("part");
            part.transform.SetParent(root.transform, false);
            var mesh = new Mesh();
            mesh.vertices = outV.GetRange(start, count).ToArray();
            mesh.uv = outT.GetRange(start, count).ToArray();
            var tris = new int[count];
            for (var i = 0; i < count; i++) tris[i] = i;
            mesh.triangles = tris;
            if (vn.Count > 0) mesh.normals = outN.GetRange(start, count).ToArray(); else mesh.RecalculateNormals();
            mesh.RecalculateBounds();
            part.AddComponent<MeshFilter>().sharedMesh = mesh;
            var r = part.AddComponent<MeshRenderer>();
            if (mat != null) r.sharedMaterial = mat;
            part.AddComponent<MeshCollider>().sharedMesh = mesh;
        }
        return root;
    }

    static int Index(string s, int count)
    {
        var i = int.Parse(s, Inv);
        return i < 0 ? count + i : i - 1;
    }

    void ApplyMaterial(GameObject go, string material)
    {
        Material m = null;
        if (material.StartsWith("tex:")) m = TextureMaterial(material.Substring(4));
        else m = Resources.FindObjectsOfTypeAll<Material>().FirstOrDefault(x => x.name == material);
        if (m == null) return;
        foreach (var r in go.GetComponentsInChildren<Renderer>(true)) r.sharedMaterial = m;
    }

    static void ApplyTint(GameObject go, Color c)
    {
        foreach (var r in go.GetComponentsInChildren<Renderer>(true))
            try
            {
                var m = r.material;
                if (m.HasProperty("_BaseColor")) m.SetColor("_BaseColor", c);
                if (m.HasProperty("_Color")) m.color = c;
            }
            catch { }
        foreach (var l in go.GetComponentsInChildren<Light>(true)) l.color = c;
    }

    readonly Dictionary<string, Material> _texMats = new Dictionary<string, Material>();

    /// <summary>Материал из своей картинки (папка textures) на основе освещённого материала игры.</summary>
    Material TextureMaterial(string file)
    {
        Material cached;
        if (_texMats.TryGetValue(file, out cached) && cached != null) return cached;
        var full = Path.Combine(Path.Combine(_root, "textures"), Path.GetFileName(file));
        if (!File.Exists(full)) return null;
        var tex = new Texture2D(2, 2);
        if (!LoadImage(tex, File.ReadAllBytes(full))) return null;
        tex.name = Path.GetFileNameWithoutExtension(file);
        var basis = BaseMaterial();
        if (basis == null) return null;
        var m = new Material(basis) { name = "tex:" + file };
        m.mainTexture = tex;
        if (m.HasProperty("_BaseMap")) m.SetTexture("_BaseMap", tex);
        _texMats[file] = m;
        return m;
    }

    /// <summary>LoadImage живёт то в Texture2D (старый Unity), то в ImageConversion (новый) — ищем отражением.</summary>
    static bool LoadImage(Texture2D tex, byte[] bytes)
    {
        foreach (var asm in AppDomain.CurrentDomain.GetAssemblies())
        {
            var conv = asm.GetType("UnityEngine.ImageConversion");
            var m = conv != null ? conv.GetMethod("LoadImage", new[] { typeof(Texture2D), typeof(byte[]) }) : null;
            if (m != null) return (bool)m.Invoke(null, new object[] { tex, bytes });
        }
        var own = typeof(Texture2D).GetMethod("LoadImage", new[] { typeof(byte[]) });
        return own != null && (bool)own.Invoke(tex, new object[] { bytes });
    }

    Light Sun()
    {
        if (RenderSettings.sun != null) return RenderSettings.sun;
        Light best = null;
        foreach (var l in FindObjectsOfType<Light>())
            if (l.type == LightType.Directional && (best == null || l.intensity > best.intensity)) best = l;
        return best;
    }

    void ApplyWorld()
    {
        var sun = Sun();
        if (sun != null)
        {
            if (_world.SunIntensity >= 0) sun.intensity = _world.SunIntensity;
            if (_world.SunPitch > -999) sun.transform.rotation = Quaternion.Euler(_world.SunPitch, _world.SunYaw > -999 ? _world.SunYaw : sun.transform.eulerAngles.y, 0);
        }
        if (_world.Ambient >= 0) RenderSettings.ambientIntensity = _world.Ambient;
        if (_world.Fog >= 0) RenderSettings.fog = _world.Fog == 1;
        if (_world.FogDensity >= 0) RenderSettings.fogDensity = _world.FogDensity;
        if (sun != null && _world.SunColor.a >= 0) sun.color = _world.SunColor;
        if (sun != null && _world.Shadow >= 0) sun.shadowStrength = _world.Shadow;
        if (_world.AmbientColor.a >= 0) { RenderSettings.ambientMode = UnityEngine.Rendering.AmbientMode.Flat; RenderSettings.ambientLight = _world.AmbientColor; }
        if (_world.FogColor.a >= 0) RenderSettings.fogColor = _world.FogColor;
        ApplyWeather();
    }

    {
        if (target.StartsWith("id:"))

    /// <summary>Поле или свойство простого типа (число, флажок, строка, перечисление, цвет, вектор), которое можно менять.</summary>
    class Member
    {
        public string Name;
        public Type Type;
        public FieldInfo Field;
        public PropertyInfo Property;
        public object Get(object o) { return Field != null ? Field.GetValue(o) : Property.GetValue(o, null); }
        public void Set(object o, object v) { if (Field != null) Field.SetValue(o, v); else Property.SetValue(o, v, null); }
    }

    static readonly Dictionary<Type, List<Member>> MemberCache = new Dictionary<Type, List<Member>>();
    static readonly HashSet<string> SkipMembers = new HashSet<string> { "name", "tag", "hideFlags", "useGUILayout", "runInEditMode", "enabled", "isActiveAndEnabled", "allowOcclusionWhenDynamic" };

    static bool Simple(Type t)
    {
        return t == typeof(float) || t == typeof(int) || t == typeof(bool) || t == typeof(string) || t == typeof(double) || t.IsEnum
            || t == typeof(Color) || t == typeof(Vector3) || t == typeof(Vector2);
    }

    /// <summary>Что показать в «Свойствах» компонента: открытые поля и свойства простых типов, без устаревших (до 40).</summary>
    static List<Member> MembersOf(Type type)
    {
        List<Member> list;
        if (MemberCache.TryGetValue(type, out list)) return list;
        list = new List<Member>();
        try
        {
            foreach (var p in type.GetProperties(BindingFlags.Public | BindingFlags.Instance))
            {
                if (!p.CanRead || !p.CanWrite || p.GetIndexParameters().Length > 0 || !Simple(p.PropertyType) || SkipMembers.Contains(p.Name)) continue;
                if (p.GetSetMethod() == null || p.IsDefined(typeof(ObsoleteAttribute), true)) continue;
                list.Add(new Member { Name = p.Name, Type = p.PropertyType, Property = p });
            }
            foreach (var f in type.GetFields(BindingFlags.Public | BindingFlags.Instance))
            {
                if (f.IsInitOnly || f.IsLiteral || !Simple(f.FieldType) || SkipMembers.Contains(f.Name) || f.IsDefined(typeof(ObsoleteAttribute), true)) continue;
                list.Add(new Member { Name = f.Name, Type = f.FieldType, Field = f });
            }
        }
        catch { }
        list = list.OrderBy(m => m.Name).Take(40).ToList();
        MemberCache[type] = list;
        return list;
    }

    static string Str(object v)
    {
        if (v == null) return "";
        if (v is float) return ((float)v).ToString("R", Inv);
        if (v is double) return ((double)v).ToString("R", Inv);
        if (v is int) return ((int)v).ToString(Inv);
        if (v is bool) return (bool)v ? "true" : "false";
        if (v is Color) return C((Color)v);
        if (v is Vector3) return V((Vector3)v);
        if (v is Vector2) { var p = (Vector2)v; return p.x.ToString("R", Inv) + "," + p.y.ToString("R", Inv); }
        return v.ToString();
    }

    static object Parse(string s, Type t)
    {
        if (t == typeof(float)) return float.Parse(s.Replace(',', '.'), NumberStyles.Float, Inv);
        if (t == typeof(double)) return double.Parse(s.Replace(',', '.'), NumberStyles.Float, Inv);
        if (t == typeof(int)) return int.Parse(s, Inv);
        if (t == typeof(bool)) return s == "true" || s == "1";
        if (t == typeof(string)) return s;
        if (t.IsEnum) return Enum.Parse(t, s);
        var f = F(s);
        if (t == typeof(Color)) return new Color(f[0], f[1], f[2], f.Length > 3 ? f[3] : 1);
        if (t == typeof(Vector3)) return new Vector3(f[0], f[1], f[2]);
        if (t == typeof(Vector2)) return new Vector2(f[0], f[1]);
        throw new NotSupportedException(t.Name);
    }

    static Component ComponentOf(GameObject go, string type)
    {
        foreach (var c in go.GetComponents<Component>()) if (c != null && c.GetType().Name == type) return c;
        return null;
    }

    bool ApplyProp(Prop pr)
    {
        try
        {
            var go = FindTarget(pr.Target);
            var c = go != null ? ComponentOf(go, pr.Component) : null;
            if (c == null) return false;
            if (pr.Member == "enabled") { SetEnabled(c, pr.Value == "true"); return true; }
            var m = MembersOf(c.GetType()).FirstOrDefault(x => x.Name == pr.Member);
            if (m == null) return false;
            m.Set(c, Parse(pr.Value, m.Type));
            return true;
        }
        catch (Exception e) { Logger.LogWarning("property " + pr.Component + "." + pr.Member + ": " + e.Message); return false; }
    }

    /// <summary>Изменить свойство компонента и запомнить правку (последняя правка того же поля заменяет прежнюю).</summary>
    void SetProp(Component c, string member, string value)
    {
        var pr = new Prop { Target = TargetOf(c.gameObject), Component = c.GetType().Name, Member = member, Value = value };
        if (!ApplyProp(pr)) return;
        _props.RemoveAll(x => x.Target == pr.Target && x.Component == pr.Component && x.Member == pr.Member);
        _props.Add(pr);
        Dirty();
    }

    static bool? EnabledOf(Component c)
    {
        if (c is Behaviour) return ((Behaviour)c).enabled;
        if (c is Renderer) return ((Renderer)c).enabled;
        if (c is Collider) return ((Collider)c).enabled;
        return null;
    }

    static void SetEnabled(Component c, bool on)
    {
        if (c is Behaviour) ((Behaviour)c).enabled = on;
        else if (c is Renderer) ((Renderer)c).enabled = on;
        else if (c is Collider) ((Collider)c).enabled = on;
    }
