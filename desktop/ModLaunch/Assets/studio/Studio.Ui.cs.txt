    bool _open;
    float _timeScale = 1;

    CursorLockMode _lock;

    string _status = "";
    float _statusAt;

    void Status(string text)
    {
        _status = text;
        _statusAt = Time.realtimeSinceStartup;
        if (_dev && !string.IsNullOrEmpty(text)) AddLog("ModScript Studio: " + text, LogType.Log);
    }

    void Open()
    {
        if (_open) return;
        if (_scene == "") { _scene = SceneManager.GetActiveScene().name; Load(); }
        _open = true;
        _timeScale = Time.timeScale;
        Time.timeScale = 0;
        _lock = Cursor.lockState;
        _cursor = Cursor.visible;
        var game = Camera.main;
        if (game == null) foreach (var c in Camera.allCameras) if (c.enabled && (game == null || c.depth > game.depth)) game = c;
        var go = new GameObject("ModScript Studio Camera");
        DontDestroyOnLoad(go);
        _cam = go.AddComponent<Camera>();
        if (game != null)
        {
            _cam.CopyFrom(game);
            go.transform.position = game.transform.position;
            go.transform.rotation = game.transform.rotation;
        }
        else go.transform.position = new Vector3(0, 10, -10);
        _cam.depth = 100;
        _cam.targetTexture = null;
        _yaw = go.transform.eulerAngles.y;
        _pitch = go.transform.eulerAngles.x;
        if (_pitch > 180) _pitch -= 360;
        _orbit = go.transform.position + go.transform.forward * 10f;
        ModScriptFollow.Target = go.transform;
        Status(T("hint", "Right mouse + WASD — fly, Alt + left mouse — orbit, click — select"));
        // Экранный интерфейс игры на время редактора прячем (свой холст UnityExplorer — нет).
        _hiddenUi.Clear();
        foreach (var c in FindObjectsOfType<Canvas>())
            if (c.enabled && c.isRootCanvas && c.renderMode == RenderMode.ScreenSpaceOverlay && !ExplorerUi(c.gameObject)) { c.enabled = false; _hiddenUi.Add(c); }
        foreach (var b in FindObjectsOfType<ModScriptBehaviour>()) b.Paused = true;
    }

    void Close()
    {
        if (!_open) return;
        if (_simulate) ToggleSimulate();
        CancelPlace();
        if (_drag >= 0) EndDrag();
        ShowAll();
        _open = false;
        Save();
        Time.timeScale = _timeScale <= 0 ? 1 : _timeScale;
        Cursor.lockState = _lock;
        Cursor.visible = _cursor;
        if (_cam != null) Destroy(_cam.gameObject);
        _cam = null;
        if (_gridGo != null) Destroy(_gridGo);
        ModScriptFollow.Target = null;
        ReleaseTarget();
        _gridGo = null;
        _selection.Clear();
        _keys.Clear();
        _rmb = _mmb = false;
        foreach (var p in _placed) if (p.Go != null) SetPhysics(p.Go, p.Physics);
        foreach (var c in _hiddenUi) if (c != null) c.enabled = true;
        _hiddenUi.Clear();
        foreach (var b in FindObjectsOfType<ModScriptBehaviour>()) { b.Rebase(); b.Paused = false; }
    }

    static bool ExplorerUi(GameObject go)
    {
        var n = go.transform.root.name;
        return n.IndexOf("UniverseLib", StringComparison.OrdinalIgnoreCase) >= 0 || n.IndexOf("Explorer", StringComparison.OrdinalIgnoreCase) >= 0;
    }

    // ------------------------------------------------------------------ камера: как в UE5

    readonly HashSet<KeyCode> _keys = new HashSet<KeyCode>();
    float _yaw, _pitch, _speed = 1f, _fps = 60;
    Vector3 _orbit;
    bool _rmb, _mmb, _orbiting, _typing;

    void Update()
    {
        if (_autotest) AutoTest();
        if (_dirty && Time.realtimeSinceStartup - _dirtyAt > 0.8f) Save();
        if (!_open || _cam == null) return;
        Cursor.lockState = CursorLockMode.None;
        Cursor.visible = true;
        var dt = Mathf.Min(Time.unscaledDeltaTime, 0.1f);
        if (dt > 0) _fps = Mathf.Lerp(_fps, 1f / dt, 0.05f);
        UpdateGrid();
        var step = Time.unscaledDeltaTime * 7f;
        _leftAnim = Mathf.MoveTowards(_leftAnim, _showToolbox ? 1 : 0, step);
        _rightAnim = Mathf.MoveTowards(_rightAnim, _rightOpen ? 1 : 0, step);
        if (_showToolbox) RenderPreviews();
        if (_typing) return;
        var t = _cam.transform;
        var move = Vector3.zero;
        // Стрелки — всегда; WASD/QE — только с зажатой правой кнопкой (без неё Q/W/E/R — инструменты).
        if (_keys.Contains(KeyCode.UpArrow)) move += t.forward;
        if (_keys.Contains(KeyCode.DownArrow)) move -= t.forward;
        if (_keys.Contains(KeyCode.RightArrow)) move += t.right;
        if (_keys.Contains(KeyCode.LeftArrow)) move -= t.right;
        if (_keys.Contains(KeyCode.PageUp)) move += Vector3.up;
        if (_keys.Contains(KeyCode.PageDown)) move -= Vector3.up;
        if (_rmb)
        {
            if (_keys.Contains(KeyCode.W)) move += _cam.orthographic ? Vector3.up : t.forward;
            if (_keys.Contains(KeyCode.S)) move -= _cam.orthographic ? Vector3.up : t.forward;
            if (_keys.Contains(KeyCode.D)) move += t.right;
            if (_keys.Contains(KeyCode.A)) move -= t.right;
            if (_keys.Contains(KeyCode.E)) move += Vector3.up;
            if (_keys.Contains(KeyCode.Q)) move -= Vector3.up;
        }
        var fast = _keys.Contains(KeyCode.LeftShift) || _keys.Contains(KeyCode.RightShift) ? 4f : 1f;
        t.position += move * 7f * _speed * fast * dt;
    }

    void LateUpdate()
    {
        if (!_open) return;
        Cursor.lockState = CursorLockMode.None;
        Cursor.visible = true;
        if (!_simulate && Time.timeScale != 0) Time.timeScale = 0;
    }

    void Look(Vector2 delta)
    {
        if (_cam.orthographic) { _cam.transform.position -= new Vector3(delta.x, -delta.y, 0) * (_cam.orthographicSize * 2 / Screen.height) * _ui; return; }
        _yaw += delta.x * 0.22f * _ui;
        _pitch = Mathf.Clamp(_pitch + delta.y * 0.22f * _ui, -89, 89);
        _cam.transform.rotation = Quaternion.Euler(_pitch, _yaw, 0);
    }

    void Orbit(Vector2 delta)
    {
        var dist = Vector3.Distance(_cam.transform.position, _orbit);
        _yaw += delta.x * 0.3f * _ui;
        _pitch = Mathf.Clamp(_pitch + delta.y * 0.3f * _ui, -89, 89);
        var rot = Quaternion.Euler(_pitch, _yaw, 0);
        _cam.transform.rotation = rot;
        _cam.transform.position = _orbit - rot * Vector3.forward * dist;
    }

    // ------------------------------------------------------------------ симуляция (как Simulate в UE5, Alt+S)

    bool _simulate;
    List<Snap> _simSnap;

    /// <summary>Время идёт, физика и поведения работают, камера — редактора; остановка — всё возвращается на место.</summary>
    void ToggleSimulate()
    {
        if (!_simulate)
        {
            if (_drag >= 0) EndDrag();
            CancelPlace();
            var gos = _placed.Where(p => p.Go != null).Select(p => p.Go).Concat(FindObjectsOfType<Rigidbody>().Select(r => r.gameObject)).Distinct().ToList();
            _simSnap = SnapOf(gos);
            _simulate = true;
            Time.timeScale = _timeScale <= 0 ? 1 : _timeScale;
            foreach (var p in _placed) if (p.Go != null) SetPhysics(p.Go, p.Physics);
            foreach (var b in FindObjectsOfType<ModScriptBehaviour>()) { b.Rebase(); b.Paused = false; }
            Status(T("simulating", "Simulation: physics and behaviours run · Alt+S — stop"));
        }
        else
        {
            _simulate = false;
            Time.timeScale = 0;
            foreach (var p in _placed) if (p.Go != null) SetPhysics(p.Go, false);
            foreach (var b in FindObjectsOfType<ModScriptBehaviour>()) b.Paused = true;
            if (_simSnap != null) Restore(_simSnap);
            _simSnap = null;
            foreach (var rb in FindObjectsOfType<Rigidbody>()) if (!rb.isKinematic) { rb.velocity = Vector3.zero; rb.angularVelocity = Vector3.zero; }
            Status(T("simStopped", "Simulation stopped — everything is back in place"));
        }
    }

    /// <summary>
    /// ModLaunch --dev-prepare … --autotest: кнопка в игре, редактор со стрелками всех трёх видов, свойства объекта уровня,
    /// освещение, установка под курсор, рамка выделения; проверка отмены/повтора, групп и свойств; отчёт и выход.
    /// </summary>
    void AutoTest()
    {
        if (_testStage == 0)
        {
            if (_testAt > 0 && Time.realtimeSinceStartup > _testAt) { _testStage = 1; _testAt = Time.realtimeSinceStartup; }
            else if (_testAt <= 0 && Time.realtimeSinceStartup > 40f) { _testStage = 1; _testAt = Time.realtimeSinceStartup; }
            return;
        }
        if (Time.realtimeSinceStartup < _testAt) return;
        try { TestStep(_testStage); }
        catch (Exception e) { Logger.LogError("self-test step " + _testStage + ": " + e); _testLog.Add("step" + _testStage + "=error " + e.Message); }
        _testStage++;
        _testAt = Time.realtimeSinceStartup + 1.6f;
    }

    void TestStep(int s)
    {
        switch (s)
        {
            case 1: Shot("autotest-1-button.png"); break;
            case 2: Open(); break;
            case 3: TestBuild(); break;
            case 4: Shot("autotest-2-move.png"); break;
            case 5: _mode = "rotate"; _local = true; break;
            case 6: Shot("autotest-3-rotate.png"); break;
            case 7: _mode = "move"; _local = false; OpenPalette(T("k.stairs", "Stairs").Split(' ')[0].Trim(':', ',')); break;
            case 8: Shot("autotest-4-palette.png"); break;
            case 9: _palette = false; _selection.Clear(); _showToolbox = true; _addCat = "nature"; RunAct(FindAct("weather.snow")); break;
            case 10: Shot("autotest-5-add.png"); break;
            case 11:
                _showToolbox = false;
                var level = Pick(new Vector2(ViewCenter.x, ViewCenter.y - _vpPx.height * 0.25f), true);
                if (level == null || PlacedOf(level) != null || level.name.StartsWith("ModScript Studio"))
                    level = FindObjectsOfType<MeshRenderer>().Where(r => r.isVisible && PlacedOf(r.gameObject) == null && !r.name.StartsWith("ModScript Studio")).Select(r => r.gameObject).FirstOrDefault();
                Select(level, false);
                if (level != null) foreach (var c in level.GetComponents<Component>()) if (c != null) _openComp.Add(c.GetType().Name);
                _rightOpen = true; _rightTab = "props"; _dockMore = true;
                _testLog.Add("levelObject=" + (level != null ? level.name : "none"));
                break;
            case 12: Shot("autotest-6-properties.png"); break;
            case 13: _rightOpen = false; _dockMore = false; _selection.Clear(); _selection.Add(_testCube); _pop = "color"; break;
            case 14: Shot("autotest-7-colors.png"); break;
            case 15: _pop = ""; TestAllCommands(); break;
            case 16: RunAct(FindAct("time.18")); RunAct(FindAct("fog.d0.02")); _selection.Clear(); _showStats = true; break;
            case 17: Shot("autotest-8-sunset.png"); break;
            case 18:
                // Симуляция: шар с физикой падает, после остановки — на прежнем месте.
                var ps = PlacedExact(_testSphere);
                if (ps != null) ps.Physics = true;
                _simY = _testSphere.transform.position.y;
                ToggleSimulate();
                break;
            case 19:
                var fell = _testSphere.transform.position.y < _simY - 0.05f;
                ToggleSimulate();
                _testLog.Add("simulate=" + (fell ? "ok" : "fail") + " restored=" + (Mathf.Abs(_testSphere.transform.position.y - _simY) < 0.01f ? "ok" : "fail"));
                break;
            case 20:
                Report();
                foreach (var p in _placed.ToArray()) { if (p.Go != null) Destroy(p.Go); _placed.Remove(p); }
                _props.Clear();
                _bookmarks.Clear();
                _world = new World();
                Close();
                break;
            case TestLast: _testDone = true; Application.Quit(); break;
        }
    }

    /// <summary>Самопроверка: фигуры, модель OBJ, поведение; отмена и повтор, группа, свойство компонента; витрина деталей.</summary>
    void TestBuild()
    {
        _testCube = Place("prim:Cube", T("shape.Cube", "Cube"));
        var b = _testCube.AddComponent<ModScriptBehaviour>();
        b.Spin = 45; b.Rebase(); b.Pause();
        _testSphere = Place("prim:Sphere", T("shape.Sphere", "Sphere"));
        _testSphere.transform.position = _testCube.transform.position + FlatRight() * 2.2f;
        var obj = Path.Combine(Path.Combine(_root, "models"), "autotest-pyramid.obj");
        File.WriteAllText(obj, "v -1 0 -1\nv 1 0 -1\nv 1 0 1\nv -1 0 1\nv 0 1.6 0\nf 1 2 5\nf 2 3 5\nf 3 4 5\nf 4 1 5\nf 1 4 3 2\n");
        var model = Place("obj:autotest-pyramid.obj", "Pyramid");
        if (model != null) model.transform.position = _testCube.transform.position - FlatRight() * 2.4f;
        try { File.Delete(obj); } catch { }
        Undo();
        var undone = model != null && !model.activeSelf;
        Redo();
        _testLog.Add("undo=" + (undone && model.activeSelf ? "ok" : "fail"));
        _selection.Clear(); _selection.Add(_testCube); _selection.Add(_testSphere);
        Group();
        var grouped = _testCube.transform.parent != null && _testCube.transform.parent == _testSphere.transform.parent;
        Undo();
        _testLog.Add("group=" + (grouped && _testCube.transform.parent == null ? "ok" : "fail"));
        var col = _testCube.GetComponent<BoxCollider>();
        EditProp(col, "isTrigger", "true");
        _testLog.Add("props=" + (col.isTrigger && _props.Any(p => p.Member == "isTrigger") ? "ok" : "fail"));
        Undo();
        // Витрина: каждая готовая деталь собирается; дюжина — рядом в кадре.
        var built = 0;
        var broken = new List<string>();
        foreach (var k in Kits())
        {
            GameObject go = null;
            try { go = Kit(k.Id); } catch (Exception e) { broken.Add(k.Id + " " + e.Message); }
            if (go == null) { if (broken.Count == 0 || !broken[broken.Count - 1].StartsWith(k.Id)) broken.Add(k.Id); continue; }
            built++;
            Destroy(go);
        }
        _testLog.Add("kits=" + built + "/" + Kits().Count + (broken.Count > 0 ? " broken: " + string.Join(", ", broken.Take(8).ToArray()) : ""));
        var showcase = new[] { "house", "tree:1.5", "pine:1.5", "car:red", "stairs:5:1", "arch:2:3", "lamppost", "fountain", "rock:1:1", "coin", "bench", "portal" };
        var basis = _testCube.transform.position - FlatRight() * 12f + Vector3.Cross(FlatRight(), Vector3.up) * -6f;
        for (var i = 0; i < showcase.Length; i++)
        {
            var go = Make("kit:" + showcase[i], basis + FlatRight() * i * 3.2f, Quaternion.identity);
            if (go == null) continue;
            var shown = Kits().FirstOrDefault(k => k.Id == showcase[i]);
            var p = Register(go, "kit:" + showcase[i], shown != null ? shown.Name : showcase[i]);
            p.Go.transform.position = basis + FlatRight() * i * 3.2f;
        }
        _testLog.Add("handleHit=" + TestHandle());
        _selection.Clear(); _selection.Add(_testSphere); _selection.Add(_testCube);
        _mode = "move";
        FocusOn(_testCube);
        _cam.transform.position += Vector3.up * 2.5f - _cam.transform.forward * 9f;
    }

    string TestHandle()
    {
        _selection.Clear(); _selection.Add(_testCube);
        _mode = "move";
        var o = _testCube.transform.position;
        var tip = Scr(o + Vector3.right * GizmoSize(o) * 0.6f);
        var h = HandleAt(tip);
        return h == 0 ? "ok" : "fail " + h;
    }

    /// <summary>Каждая команда — на свежем кубике; затем отменить всё. Считаем, сколько прошли без ошибок.</summary>
    void TestAllCommands()
    {
        var skip = new HashSet<string> { "clearAll", "view.ui", "file.revert", "file.models", "file.textures", "file.folder", "sel.none", "file.backup", "file.export" };
        var acts = Acts().Where(a => !skip.Contains(a.Id) && !a.Id.StartsWith("shot.") && !a.Id.StartsWith("set.") && !a.Id.StartsWith("phys.kick") && a.Id != "phys.push" && a.Id != "phys.explode" && !a.Id.StartsWith("accent.") && !a.Id.StartsWith("uiScale.")).ToList();
        var failed = new List<string>();
        var undoBefore = _undo.Count;
        var cam = _cam.transform.position;
        var camRot = _cam.transform.rotation;
        var fov = _cam.fieldOfView;
        foreach (var a in acts)
        {
            if (_testCube == null || !_testCube.activeInHierarchy) { _testCube = Place("prim:Cube", "Cube"); }
            _selection.Clear();
            _selection.Add(_testCube);
            if (a.Id.StartsWith("info.distance") || a.Id.StartsWith("align") || a.Id.StartsWith("distribute") || a.Id == "stackOn" || a.Id == "lookAtLast") _selection.Insert(0, _testSphere);
            try { a.Run(); }
            catch (Exception e) { failed.Add(a.Id + ": " + e.GetType().Name); }
            CancelPlace();
            if (_simulate) ToggleSimulate();
            _palette = false;
        }
        var undoErrors = 0;
        while (_undo.Count > undoBefore)
        {
            var c = _undo[_undo.Count - 1];
            _undo.RemoveAt(_undo.Count - 1);
            try { c.Undo(); } catch { undoErrors++; }
        }
        _cam.transform.position = cam;
        _cam.transform.rotation = camRot;
        _cam.fieldOfView = fov;
        _cam.orthographic = false;
        _testLog.Add("commands=" + Acts().Count + " ran=" + acts.Count + " failed=" + failed.Count + (failed.Count > 0 ? " (" + string.Join("; ", failed.Take(10).ToArray()) + ")" : "") + " undoErrors=" + undoErrors);
        _testLog.Add("placedAfterUndo=" + _placed.Count);
    }

    void Shot(string name)
    {
        var path = Path.Combine(_root, name);
        var sc = Type.GetType("UnityEngine.ScreenCapture, UnityEngine.ScreenCaptureModule") ?? Type.GetType("UnityEngine.ScreenCapture, UnityEngine");
        var m = sc != null ? sc.GetMethod("CaptureScreenshot", new[] { typeof(string) }) : null;
        if (m != null) m.Invoke(null, new object[] { path });
        else Logger.LogWarning("no ScreenCapture in this game");
    }

    void Report()
    {
        var pipeline = Type.GetType("UnityEngine.Rendering.GraphicsSettings, UnityEngine.CoreModule");
        object rp = null;
        try { var prop = pipeline != null ? pipeline.GetProperty("currentRenderPipeline") ?? pipeline.GetProperty("renderPipelineAsset") : null; rp = prop != null ? prop.GetValue(null, null) : null; } catch { }
        var lines = new List<string>
        {
            "unity=" + Application.unityVersion,
            "screen=" + Screen.width + "x" + Screen.height + " ui=" + _ui.ToString("0.00", Inv) + " viewport=" + _vpPx,
            "font=" + (_font != null ? _font.name : "none") + " fontOk=" + _fontOk,
            "pipeline=" + (rp != null ? rp.GetType().Name : "builtin"),
            "scene=" + _scene,
            "library=" + (_library != null ? _library.Count : -1) + " materials=" + (_materials != null ? _materials.Count : -1),
            "placed=" + _placed.Count + " selected=" + _selection.Count,
            "grid=" + (_gridGo != null && _gridGo.activeSelf) + " gridRendered=" + (_gridGo != null && _gridGo.GetComponent<MeshRenderer>().isVisible),
            "base=" + (BaseMaterial() != null ? BaseMaterial().shader.name : "none"),
            "log=" + _log.Count + " previews=" + _previews.Count + " weather=" + (_world.Weather ?? "") + " profile=" + _gameName + "/" + _preset + " quick=" + QuickList().Count + "/" + _profQuick.Length,
        };
        lines.AddRange(_testLog);
        File.WriteAllLines(Path.Combine(_root, "autotest.txt"), lines.ToArray(), Encoding.UTF8);
        Logger.LogInfo("ModScript Studio self-test: " + string.Join("; ", lines.ToArray()));
    }

    /// <summary>Скруглённая плашка для 9-срезов (сглаженные углы, необязательная рамка).</summary>
    static Texture2D Rounded(Color fill, int r, Color edge, float edgeWidth)
    {
        var size = r * 2 + 2;
        var t = new Texture2D(size, size, TextureFormat.RGBA32, false) { hideFlags = HideFlags.HideAndDontSave, filterMode = FilterMode.Bilinear, wrapMode = TextureWrapMode.Clamp };
        for (var y = 0; y < size; y++)
            for (var x = 0; x < size; x++)
            {
                var cx = Mathf.Clamp(x + 0.5f, r, size - r);
                var cy = Mathf.Clamp(y + 0.5f, r, size - r);
                var d = Mathf.Sqrt((x + 0.5f - cx) * (x + 0.5f - cx) + (y + 0.5f - cy) * (y + 0.5f - cy));
                var inside = Mathf.Clamp01(r - d + 0.5f);
                var border = edgeWidth > 0 ? Mathf.Clamp01(d - (r - edgeWidth) + 0.5f) : 0;
                var c = Color.Lerp(fill, edge, border);
                c.a *= inside;
                t.SetPixel(x, y, c);
            }
        t.Apply();
        return t;
    }

    readonly Dictionary<string, Texture2D> _icons = new Dictionary<string, Texture2D>();

    Texture2D Icon(string name)
    {
        Texture2D t;
        if (_icons.TryGetValue(name, out t) && t != null) return t;
        var ops = IconOps(name);
        const int S = 48;
        const float k = 2f;
        t = new Texture2D(S, S, TextureFormat.RGBA32, false) { hideFlags = HideFlags.HideAndDontSave, filterMode = FilterMode.Bilinear, wrapMode = TextureWrapMode.Clamp };
        var px = new Color[S * S];
        for (var y = 0; y < S; y++)
            for (var x = 0; x < S; x++)
            {
                var p = new Vector2((x + 0.5f) / k, (S - y - 0.5f) / k);
                var cov = 0f;
                foreach (var op in ops) cov = Mathf.Max(cov, Coverage(op, p, k));
                px[y * S + x] = new Color(1, 1, 1, cov);
            }
        t.SetPixels(px);
        t.Apply();
        _icons[name] = t;
        return t;
    }

    static float SegDist(Vector2 p, Vector2 a, Vector2 b)
    {
        var ab = b - a;
        var t = ab.sqrMagnitude < 1e-6f ? 0 : Mathf.Clamp01(Vector2.Dot(p - a, ab) / ab.sqrMagnitude);
        return Vector2.Distance(p, a + ab * t);
    }

    static float Cross2(Vector2 a, Vector2 b) { return a.x * b.y - a.y * b.x; }

    static float Coverage(float[] o, Vector2 p, float k)
    {
        float d;
        switch ((int)o[0])
        {
            case 0: d = SegDist(p, new Vector2(o[1], o[2]), new Vector2(o[3], o[4])) - o[5] * 0.5f; break;
            case 1: d = Mathf.Abs(Vector2.Distance(p, new Vector2(o[1], o[2])) - o[3]) - o[4] * 0.5f; break;
            case 2:
                {
                    var c = new Vector2(o[1], o[2]);
                    var ang = Mathf.Atan2(p.y - c.y, p.x - c.x) * Mathf.Rad2Deg;
                    var rel = Mathf.Repeat(ang - o[4], 360f);
                    if (rel <= o[5] - o[4]) d = Mathf.Abs(Vector2.Distance(p, c) - o[3]) - o[6] * 0.5f;
                    else
                    {
                        var a0 = o[4] * Mathf.Deg2Rad; var a1 = o[5] * Mathf.Deg2Rad;
                        var e0 = c + new Vector2(Mathf.Cos(a0), Mathf.Sin(a0)) * o[3];
                        var e1 = c + new Vector2(Mathf.Cos(a1), Mathf.Sin(a1)) * o[3];
                        d = Mathf.Min(Vector2.Distance(p, e0), Vector2.Distance(p, e1)) - o[6] * 0.5f;
                    }
                    break;
                }
            case 3: d = Vector2.Distance(p, new Vector2(o[1], o[2])) - o[3]; break;
            case 4:
            case 5:
                {
                    var half = new Vector2(o[3], o[4]) * 0.5f;
                    var c = new Vector2(o[1], o[2]) + half;
                    var q = new Vector2(Mathf.Abs(p.x - c.x), Mathf.Abs(p.y - c.y)) - half + new Vector2(o[5], o[5]);
                    var sd = new Vector2(Mathf.Max(q.x, 0), Mathf.Max(q.y, 0)).magnitude + Mathf.Min(Mathf.Max(q.x, q.y), 0) - o[5];
                    d = (int)o[0] == 4 ? sd : Mathf.Abs(sd) - o[6] * 0.5f;
                    break;
                }
            default:
                {
                    var a = new Vector2(o[1], o[2]); var b = new Vector2(o[3], o[4]); var c = new Vector2(o[5], o[6]);
                    var dist = Mathf.Min(SegDist(p, a, b), Mathf.Min(SegDist(p, b, c), SegDist(p, c, a)));
                    var s1 = Cross2(b - a, p - a); var s2 = Cross2(c - b, p - b); var s3 = Cross2(a - c, p - c);
                    var inside = (s1 >= 0 && s2 >= 0 && s3 >= 0) || (s1 <= 0 && s2 <= 0 && s3 <= 0);
                    d = inside ? -dist : dist;
                    break;
                }
        }
        return Mathf.Clamp01(0.5f - d * k);
    }

    static float[] IL(float a, float b, float c, float d, float w) { return new[] { 0f, a, b, c, d, w }; }

    static float[] IH(float x, float y, float deg, float s)
    {
        var a = deg * Mathf.Deg2Rad;
        var d = new Vector2(Mathf.Cos(a), Mathf.Sin(a));
        var n = new Vector2(-d.y, d.x);
        var b = new Vector2(x, y) - d * s;
        var p1 = b + n * s * 0.62f;
        var p2 = b - n * s * 0.62f;
        return IT(x, y, p1.x, p1.y, p2.x, p2.y);
    }

    static float[] IArcHead(float cx, float cy, float r, float deg, float s)
    {
        var a = deg * Mathf.Deg2Rad;
        var end = new Vector2(cx + r * Mathf.Cos(a), cy + r * Mathf.Sin(a));
        var t = new Vector2(-Mathf.Sin(a), Mathf.Cos(a));
        var tip = end + t * s * 0.75f;
        return IH(tip.x, tip.y, Mathf.Atan2(t.y, t.x) * Mathf.Rad2Deg, s);
    }

    /// <summary>
    /// Камера редактора рисует в свою текстуру размером с окно просмотра, а окно выводит её между панелями. Так вид не
    /// зависит от того, как конвейер игры (встроенный, URP, HDRP) обходится с частью экрана, и стрелки совпадают с объектами.
    /// </summary>
    void EnsureTarget()
    {
        if (_cam == null) return;
        var w = Mathf.Max(64, (int)_vpPx.width);
        var h = Mathf.Max(64, (int)_vpPx.height);
        if (_rt != null && _rt.width == w && _rt.height == h && _cam.targetTexture == _rt) return;
        ReleaseTarget();
        _rt = new RenderTexture(w, h, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.sRGB) { name = "ModScript Studio View", hideFlags = HideFlags.HideAndDontSave };
        _rt.Create();
        _cam.rect = new Rect(0, 0, 1, 1);
        _cam.targetTexture = _rt;
        _cam.ResetAspect();
    }

    void ReleaseTarget()
    {
        if (_cam != null) _cam.targetTexture = null;
        if (_rt == null) return;
        _rt.Release();
        Destroy(_rt);
        _rt = null;
    }

        var thumbW = GUI.skin.verticalScrollbarThumb.fixedWidth;
        GUI.skin.font = _font;

            var h = Screen.height / _ui;

    void Editor(float w, float h, Event e)
    {
        LayoutRects(w, h);
        _typing = GUIUtility.keyboardControl != 0;
        if (!_typing) _edit.Clear();
        if (e.type == EventType.Layout) { _uiLast = new List<Rect>(_uiNow); _uiNow.Clear(); }
        var mouse = e.mousePosition;
        var inView = _vpRect.Contains(mouse) && !OverUi(mouse);
        // Точка мыши в окне просмотра — в точках его текстуры (снизу вверх), как у камеры редактора.
        var screen = new Vector2(mouse.x * _ui - _vpPx.x, Screen.height - mouse.y * _ui - _vpPx.y);
        // Клавиши палитры — до её поля ввода (иначе стрелки и Enter заберёт поле).
        if (_palette && e.type == EventType.KeyDown) PaletteKeys(e);

        if (e.type == EventType.Repaint)
        {
            // Наведённое: стрелка или объект — подсвечиваются; объект из библиотеки следует за курсором.
            if (!_autotest)
            {
                _hover = inView && _drag < 0 && _ghost == null ? HandleAt(screen) : -1;
                _hoverGo = null;
                if (inView && _hover < 0 && _drag < 0 && !_marquee && _ghost == null)
                {
                    RaycastHit hit;
                    if (!_cam.orthographic && Physics.Raycast(_cam.ScreenPointToRay(new Vector3(screen.x, screen.y, 0)), out hit, 5000f, ~0, QueryTriggerInteraction.Ignore))
                    {
                        var p = TopPlaced(hit.collider.gameObject);
                        var r = hit.collider.GetComponentInParent<Renderer>();
                        _hoverGo = p != null ? p.Go : (r != null ? r.gameObject : hit.collider.gameObject);
                    }
                }
                if (inView && _ghost != null) MoveGhost(screen);
            }
            // Сначала стрелки и рамки — в текстуру вида, потом сама текстура — на весь экран.
            DrawViewport();
            if (_rt != null) GUI.DrawTexture(_vpRect, _rt, ScaleMode.StretchToFill, false);
            ViewportOverlay();
        }

        if (_clean)
        {
            // Чистый вид: только кнопка вернуть интерфейс.
            if (GUI.Button(Reg(new Rect(w - 200, 14, 186, 34)), T("showUi", "Show interface  F10"), _pill)) Later(() => _clean = false);
        }
        else
        {
            TopIslands(w);
            if (_showStats) StatsIsland(w);
            if (_leftRect.width > 0) AddDrawer();
            if (_rightRect.width > 0) RightDrawer();
            if (_showOutput) OutputPanel();
            ContextBar(w, h);
            Dock(w, h);
            DockMore();
            Toast(w, h);
            if (_showHelp) HelpCard(w, h);
            if (_palette) Palette(w, h);
        }

        // Мышь в окне просмотра — после интерфейса (его кнопки уже обработали свои щелчки).
        if (e.type == EventType.MouseUp)
        {
            if (e.button == 1) _rmb = false;
            if (e.button == 2) _mmb = false;
        }
        if (!_palette && (inView || _drag >= 0 || _marquee || _orbiting || _pressed || _rmb || _mmb)) ViewportMouse(e, screen, inView);
        if (_palette && e.type == EventType.MouseDown && !_palRect.Contains(mouse)) { Later(() => _palette = false); e.Use(); }
        Keyboard(e);
        if (!string.IsNullOrEmpty(GUI.tooltip) && e.type == EventType.Repaint)
        {
            var c = new GUIContent(GUI.tooltip);
            var size = _tip.CalcSize(c);
            var r = new Rect(Mathf.Min(mouse.x + 14, w - size.x - 6), Mathf.Min(mouse.y + 20, h - size.y - 4), size.x, size.y);
            GUI.Label(r, c, _tip);
        }
    }

    bool _pressed, _surface, _pressAdd, _clickInner;
    Vector2 _press;
    GameObject _pressGo;
    readonly List<Collider> _surfaceCols = new List<Collider>();
    float _surfaceLift;

    void ViewportMouse(Event e, Vector2 screen, bool inView)
    {
        switch (e.type)
        {
            case EventType.MouseDown:
                GUIUtility.keyboardControl = 0;
                if (!inView) return;
                if (e.button == 1)
                {
                    if (_ghost != null) { CancelPlace(); Status(T("cancelled", "Cancelled")); }
                    else _rmb = true;
                    e.Use();
                    return;
                }
                if (e.button == 2) { _mmb = true; e.Use(); return; }
                if (e.button != 0) return;
                if (_ghost != null) { DropGhost(e.shift); e.Use(); return; }
                if (_mode == "paint") { var pg = Pick(screen, false); if (pg != null) PaintOn(pg); e.Use(); return; }
                if (e.alt)
                {
                    _orbiting = true;
                    _orbit = _sel != null ? BoundsOf(_sel).center : _cam.transform.position + _cam.transform.forward * 10f;
                    e.Use();
                    return;
                }
                var h = HandleAt(screen);
                if (h >= 0) { BeginDrag(h, screen); e.Use(); return; }
                _pressed = true;
                _press = screen;
                _pressAdd = e.shift || e.control;
                _clickInner = e.clickCount > 1;
                _pressGo = Pick(screen, _clickInner);
                if (_pressGo != null && (_pressAdd || !_selection.Contains(_pressGo))) Select(_pressGo, _pressAdd);
                e.Use();
                return;
            case EventType.MouseDrag:
                if (e.button == 1 && _rmb) { Look(e.delta); e.Use(); return; }
                if (e.button == 2 && _mmb)
                {
                    var k = _cam.orthographic ? _cam.orthographicSize * 2 / Screen.height : 0.0025f * Mathf.Max(2f, Vector3.Distance(_cam.transform.position, _orbit));
                    _cam.transform.position -= (_cam.transform.right * e.delta.x - _cam.transform.up * e.delta.y) * k * _ui;
                    e.Use();
                    return;
                }
                if (e.button != 0) return;
                if (_orbiting) { Orbit(e.delta); e.Use(); return; }
                if (_drag >= 0) { DragTo(screen); e.Use(); return; }
                if (!_pressed) return;
                if (_surface) { SurfaceDrag(screen); e.Use(); return; }
                if (_marquee) { _marq1 = screen; e.Use(); return; }
                if ((screen - _press).magnitude > 5f * _ui)
                {
                    if (_pressGo != null && _selection.Contains(_pressGo)) BeginSurface(screen);
                    else { _marquee = true; _marq0 = _press; _marq1 = screen; }
                }
                e.Use();
                return;
            case EventType.MouseUp:
                if (e.button != 0) return;
                if (_orbiting) { _orbiting = false; e.Use(); return; }
                if (_drag >= 0) { EndDrag(); e.Use(); return; }
                if (!_pressed) return;
                _pressed = false;
                if (_surface) EndSurface();
                else if (_marquee) { _marquee = false; FinishMarquee(_pressAdd); }
                else if (_pressGo == null && !_pressAdd) Select(null, false);
                else if (_pressGo != null && !_pressAdd && _selection.Count > 1) Select(_pressGo, false);
                e.Use();
                return;
            case EventType.ScrollWheel:
                if (!inView) return;
                if (_rmb) { _speed = Mathf.Clamp(_speed * (e.delta.y > 0 ? 0.8f : 1.25f), 0.05f, 20f); Status(T("camSpeed", "Camera speed") + ": ×" + _speed.ToString("0.##", Inv)); }
                else if (_cam.orthographic) _cam.orthographicSize = Mathf.Clamp(_cam.orthographicSize * (1 + e.delta.y * 0.05f), 0.5f, 500);
                else
                {
                    var ray = _cam.ScreenPointToRay(new Vector3(screen.x, screen.y, 0));
                    _cam.transform.position -= ray.direction * e.delta.y * 0.8f * _speed;
                }
                e.Use();
                return;
        }
    }

    /// <summary>Перетаскивание самого объекта (не стрелок): он едет по поверхностям под курсором, как в Roblox Studio.</summary>
    void BeginSurface(Vector2 screen)
    {
        _surface = true;
        _before = SnapOf(_selection);
        _pos0 = _selection.Select(g => g.transform.position).ToArray();
        _surfaceCols.Clear();
        foreach (var g in _selection)
            foreach (var c in g.GetComponentsInChildren<Collider>())
                if (c.enabled) { c.enabled = false; _surfaceCols.Add(c); }
        var b = BoundsOf(_pressGo);
        _surfaceLift = _pressGo.transform.position.y - b.min.y;
        SurfaceDrag(screen);
    }

    void SurfaceDrag(Vector2 screen)
    {
        var ray = _cam.ScreenPointToRay(new Vector3(screen.x, screen.y, 0));
        RaycastHit h;
        Vector3 p;
        if (!_cam.orthographic && Physics.Raycast(ray, out h, 2000f, ~0, QueryTriggerInteraction.Ignore)) p = h.point;
        else return;
        if (_snap) { p.x = Mathf.Round(p.x / _grid) * _grid; p.z = Mathf.Round(p.z / _grid) * _grid; }
        var i0 = _selection.IndexOf(_pressGo);
        if (i0 < 0 || _pos0.Length != _selection.Count) return;
        var delta = p + Vector3.up * _surfaceLift - _pos0[i0];
        for (var i = 0; i < _selection.Count; i++) _selection[i].transform.position = _pos0[i] + delta;
    }

    void EndSurface()
    {
        _surface = false;
        foreach (var c in _surfaceCols) if (c != null) c.enabled = true;
        _surfaceCols.Clear();
        foreach (var g in _selection) Touch(g);
        PushSnap(T("move", "Move"), _before, SnapOf(_selection));
        Dirty();
    }

    void Keyboard(Event e)
    {
        if (e.type == EventType.KeyUp) { _keys.Remove(e.keyCode); return; }
        if (e.type != EventType.KeyDown || e.keyCode == KeyCode.None) return;
        if (e.keyCode == KeyCode.F8) { Later(Close); e.Use(); return; }
        if (_palette) return;
        if (_typing)
        {
            if (e.keyCode == KeyCode.Return || e.keyCode == KeyCode.KeypadEnter || e.keyCode == KeyCode.Escape) { GUIUtility.keyboardControl = 0; e.Use(); }
            return;
        }
        _keys.Add(e.keyCode);
        var k = e.keyCode;
        var ctrl = e.control || e.command;
        if (ctrl)
        {
            if (k == KeyCode.Z && e.shift) Later(Redo);
            else if (k == KeyCode.Z) Later(Undo);
            else if (k == KeyCode.Y) Later(Redo);
            else if (k == KeyCode.D) Later(Duplicate);
            else if (k == KeyCode.C) Later(Copy);
            else if (k == KeyCode.V) Later(Paste);
            else if (k == KeyCode.G && e.shift) Later(Ungroup);
            else if (k == KeyCode.G) Later(Group);
            else if (k == KeyCode.S) Later(Save);
            else if (k == KeyCode.K) Later(() => OpenPalette(""));
            else if (k == KeyCode.A) Later(() => SelectWhere(p => true));
            else if (e.shift && k >= KeyCode.Alpha1 && k <= KeyCode.Alpha9) { var n = k - KeyCode.Alpha0; Later(() => RunAct(FindAct("bm.save" + n))); }
            e.Use();
            return;
        }
        if (e.alt && k == KeyCode.S) { Later(ToggleSimulate); e.Use(); return; }
        if (e.shift && k >= KeyCode.Alpha1 && k <= KeyCode.Alpha9) { var n = k - KeyCode.Alpha0; Later(() => GoBookmark(n)); e.Use(); return; }
        if (_rmb) return;
        switch (k)
        {
            case KeyCode.Q: _mode = "select"; break;
            case KeyCode.W: case KeyCode.Alpha1: _mode = "move"; break;
            case KeyCode.E: case KeyCode.Alpha2: _mode = "rotate"; break;
            case KeyCode.R: case KeyCode.Alpha3: _mode = "scale"; break;
            case KeyCode.Space: Later(() => OpenPalette("")); break;
            case KeyCode.Tab: Later(() => _showToolbox = !_showToolbox); break;
            case KeyCode.I: Later(() => OpenRight("props")); break;
            case KeyCode.F1: Later(() => _showHelp = !_showHelp); break;
            case KeyCode.F10: Later(() => _clean = !_clean); break;
            case KeyCode.G: _snap = !_snap; break;
            case KeyCode.F: if (_sel != null) FocusOn(_sel); break;
            case KeyCode.End: Later(DropToGround); break;
            case KeyCode.Delete: Later(Delete); break;
            case KeyCode.H: if (e.alt) ShowAll(); else foreach (var g in TopSelection()) ToggleEye(g); break;
            case KeyCode.Escape:
                if (_pop != "") _pop = "";
                else if (_showHelp) _showHelp = false;
                else if (_ghost != null) CancelPlace();
                else if (_marquee) { _marquee = false; _pressed = false; }
                else Later(() => Select(null, false));
                break;
            default: return;
        }
        e.Use();
    }

    Color SwatchOf(Act a)
    {
        if (!a.Id.StartsWith("color.")) return Color.clear;

        for (var i = 0; i < Neutrals.Length; i++) if (key == Neutrals[i]) return NeutralC[i];

    }

    List<Act> PaletteResults()
    {
        var all = Acts();
        var q = _pq.Trim().ToLowerInvariant();
        if (q == "")
        {
            if (_pGroup != "") return all.Where(a => a.Group == _pGroup).ToList();
            var list = _recent.Select(FindAct).Where(a => a != null).ToList();
            list.AddRange(all.Where(a => a.Group != "add" && !list.Contains(a)).Take(60));
            return list;
        }
        var words = q.Split(new[] { " " }, StringSplitOptions.RemoveEmptyEntries);
        return all.Where(a => (_pGroup == "" || a.Group == _pGroup) && words.All(wd => a.Search.Contains(wd)))
            .OrderBy(a => a.Name.ToLowerInvariant().StartsWith(words[0]) ? 0 : 1).ThenBy(a => a.Name.Length).Take(120).ToList();
    }

    void PaletteKeys(Event e)
    {
        switch (e.keyCode)
        {
            case KeyCode.Escape: Later(() => _palette = false); e.Use(); break;
            case KeyCode.DownArrow: _pIndex = Math.Min(_pIndex + 1, Math.Max(0, _pList.Count - 1)); EnsureVisible(); e.Use(); break;
            case KeyCode.UpArrow: _pIndex = Math.Max(_pIndex - 1, 0); EnsureVisible(); e.Use(); break;
            case KeyCode.Return:
            case KeyCode.KeypadEnter:
                if (_pIndex < _pList.Count) { var a = _pList[_pIndex]; var keep = e.shift; Later(() => { if (!keep) _palette = false; RunAct(a); }); }
                e.Use();
                break;
        }
    }

    int TreeRow(int depth, string icon, string label, bool on, bool kids, bool open, GameObject eyeOf, Color iconColor)
    {
        var r = GUILayoutUtility.GetRect(10, 24, GUILayout.ExpandWidth(true), GUILayout.Height(24));
        var e = Event.current;
        var hover = r.Contains(e.mousePosition);
        var x = r.x + 6 + depth * 14;
        var chev = new Rect(x, r.y + 4, 16, 16);
        var eye = new Rect(r.xMax - 26, r.y + 3, 18, 18);
        var hidden = eyeOf != null && _eyeHidden.ContainsKey(eyeOf);
        if (e.type == EventType.Repaint)
        {
            if (on) { Fill(r, AmberSoft); Fill(new Rect(r.x, r.y, 2, r.height), Amber); }
            else if (hover) Fill(r, new Color(1, 1, 1, 0.045f));
            if (kids) DrawIcon(chev, open ? "chevD" : "chevR", Muted);
            DrawIcon(new Rect(x + 18, r.y + 4, 16, 16), icon, hidden ? Faint : iconColor);
            GUI.Label(new Rect(x + 40, r.y, r.xMax - x - 70, r.height), label, on ? _bold : (hidden ? _small : _txt));
            if (eyeOf != null && (hover || hidden)) DrawIcon(eye, hidden ? "eyeOff" : "eye", hidden ? Muted : Faint);
        }
        if (e.type == EventType.MouseDown && e.button == 0 && hover)
        {
            e.Use();
            if (kids && chev.Contains(e.mousePosition)) return 2;
            if (eyeOf != null && eye.Contains(e.mousePosition)) return 3;
            return e.clickCount > 1 ? 4 : 1;
        }
        return 0;
    }

    void PlacedNode(Transform t, int depth)
    {
        var go = t.gameObject;
        var p = PlacedExact(go);
        if (p == null || !go.activeSelf) return;
        var kids = new List<Transform>();
        foreach (Transform c in t) if (PlacedExact(c.gameObject) != null && c.gameObject.activeSelf) kids.Add(c);
        if (_expSearch != "" && go.name.IndexOf(_expSearch, StringComparison.OrdinalIgnoreCase) < 0 && kids.Count == 0) return;
        var id = go.GetInstanceID();
        var open = _expanded.Contains(id) || _expSearch != "";
        var beh = go.GetComponent<ModScriptBehaviour>();
        var label = go.name + (beh != null && beh.Code() != "" ? "  ⚙" : "");
        var res = TreeRow(depth, IconOfSource(p.Source), label, _selection.Contains(go), kids.Count > 0, open, go, Ink);
        Explore(res, go, id);
        if (open) foreach (var c in kids) PlacedNode(c, depth + 1);
    }

    static string IconOfSource(string source)
    {
        if (source.StartsWith("prim:Light")) return "bulb";
        if (source.StartsWith("prim:Spot")) return "spot";
        if (source.StartsWith("prim:Empty")) return "group";
        if (source.StartsWith("prim:Sphere")) return "sphere";
        if (source.StartsWith("prim:Cylinder")) return "cylinder";
        if (source.StartsWith("prim:Capsule")) return "capsule";
        if (source.StartsWith("prim:Plane") || source.StartsWith("prim:Quad")) return "plane";
        if (source.StartsWith("prim:")) return "cube";
        if (source.StartsWith("obj:")) return "model";
        return "model";
    }

    void Explore(int res, GameObject go, int id)
    {
        if (res == 1) { var add = Event.current.shift || Event.current.control; Later(() => Select(go, add)); }
        else if (res == 4) Later(() => { Select(go, false); FocusOn(go); });
        else if (res == 2) Later(() => { if (!_expanded.Remove(id)) _expanded.Add(id); });
        else if (res == 3) Later(() => ToggleEye(go));
    }

    void LevelNode(Transform t, int depth, ref int shown)
    {
        if (_expSearch != "" && !Matches(t, 4)) return;
        shown++;
        var id = t.GetInstanceID();
        var open = _expanded.Contains(id) || (_expSearch != "" && depth < 5);
        var go = t.gameObject;
        var icon = go.GetComponent<Light>() != null ? "bulb" : go.GetComponent<Renderer>() != null ? "cube" : t.childCount > 0 ? "folder" : "empty";
        var res = TreeRow(depth, icon, (go.activeSelf ? "" : "× ") + t.name, _selection.Contains(go), t.childCount > 0, open, go, go.activeSelf ? Muted : Faint);
        Explore(res, go, id);
        if (!open) return;
        for (var i = 0; i < t.childCount && shown < 400; i++) LevelNode(t.GetChild(i), depth + 1, ref shown);
    }

    bool Matches(Transform t, int depth)
    {
        if (t.name.IndexOf(_expSearch, StringComparison.OrdinalIgnoreCase) >= 0) return true;
        if (depth <= 0) return false;
        for (var i = 0; i < t.childCount; i++) if (Matches(t.GetChild(i), depth - 1)) return true;
        return false;
    }

    void TransformRows()
    {
        var t = _sel.transform;
        var e = Event.current;
        if (_fieldsFor != _sel || GUIUtility.keyboardControl == 0)
        {
            _fieldsFor = _sel;
            var eu = t.eulerAngles;
            _fields = new[] { t.position.x, t.position.y, t.position.z, eu.x, eu.y, eu.z, t.localScale.x, t.localScale.y, t.localScale.z }.Select(v => v.ToString("0.###", Inv)).ToArray();
        }
        var labels = new[] { T("position", "Position"), T("rotation", "Rotation"), T("scale", "Scale") };
        var changed = false;
        var fw = (_propsRect.width - 120) / 3f - 18;
        for (var row = 0; row < 3; row++)
        {
            GUILayout.BeginHorizontal();
            GUILayout.Label(labels[row], _propLabel, GUILayout.Width(76), GUILayout.Height(26));
            for (var c = 0; c < 3; c++)
            {
                var i = row * 3 + c;
                // Буква оси: тянуть мышью — менять значение (как в UE5).
                var lr = GUILayoutUtility.GetRect(16, 26, GUILayout.Width(16), GUILayout.Height(26));
                if (e.type == EventType.Repaint)
                {
                    Fill(new Rect(lr.x + 2, lr.y + 4, 14, 18), new Color(AxisColors[c].r, AxisColors[c].g, AxisColors[c].b, _scrub == i ? 1f : 0.85f));
                    GUI.Label(new Rect(lr.x + 2, lr.y + 4, 14, 18), c == 0 ? "X" : c == 1 ? "Y" : "Z", Text(Color.white, 11, FontStyle.Bold, TextAnchor.MiddleCenter));
                }
                if (e.type == EventType.MouseDown && e.button == 0 && lr.Contains(e.mousePosition))
                {
                    _scrub = i; _scrubMouse = e.mousePosition; _scrubBefore = SnapOf(_selection);
                    _scrubStart = float.Parse(_fields[i], NumberStyles.Float, Inv);
                    GUIUtility.keyboardControl = 0;
                    e.Use();
                }
                var v = GUILayout.TextField(_fields[i], _field, GUILayout.Width(fw), GUILayout.Height(24));
                if (v != _fields[i]) { _fields[i] = v; changed = true; }
            }
            GUILayout.EndHorizontal();
        }
        if (_scrub >= 0 && e.type == EventType.MouseDrag)
        {
            var step = _scrub < 3 ? 0.02f : _scrub < 6 ? 0.5f : 0.01f;
            var nv = _scrubStart + (e.mousePosition.x - _scrubMouse.x) * step;
            if (_snap && _scrub < 3) nv = Mathf.Round(nv / _grid) * _grid;
            if (_snap && _scrub >= 3 && _scrub < 6) nv = Mathf.Round(nv / AngleSnap) * AngleSnap;
            _fields[_scrub] = nv.ToString("0.###", Inv);
            ApplyFields();
            e.Use();
        }
        if (_scrub >= 0 && e.type == EventType.MouseUp)
        {
            _scrub = -1;
            PushSnap(T("section.transform", "Transform"), _scrubBefore, SnapOf(_selection));
            e.Use();
        }
        if (changed)
        {
            if (_fieldBefore == null) _fieldBefore = SnapOf(new[] { _sel });
            ApplyFields();
        }
    }

    void ApplyFields()
    {
        var nv = new float[9];
        for (var i = 0; i < 9; i++)
            if (!float.TryParse(_fields[i].Replace(',', '.'), NumberStyles.Float, Inv, out nv[i])) return;
        var t = _sel.transform;
        t.position = new Vector3(nv[0], nv[1], nv[2]);
        t.eulerAngles = new Vector3(nv[3], nv[4], nv[5]);
        t.localScale = new Vector3(nv[6], nv[7], nv[8]);
        Touch(_sel);
        Dirty();
    }

    static string Nice(string name)
    {
        var sb = new StringBuilder();
        for (var i = 0; i < name.Length; i++)
        {
            var ch = name[i];
            if (i == 0) { sb.Append(char.ToUpperInvariant(ch)); continue; }
            if (char.IsUpper(ch) && !char.IsUpper(name[i - 1])) sb.Append(' ');
            sb.Append(ch);
        }
        return sb.ToString().TrimStart('_', 'm', ' ');
    }

    static string Hex(Color c)
    {
        return "#" + ((int)(Mathf.Clamp01(c.r) * 255)).ToString("X2") + ((int)(Mathf.Clamp01(c.g) * 255)).ToString("X2") + ((int)(Mathf.Clamp01(c.b) * 255)).ToString("X2");
    }

    static bool FromHex(string s, float alpha, out Color c)
    {
        c = Color.white;
        s = s.Trim().TrimStart('#');
        if (s.Length != 6) return false;
        int v;
        if (!int.TryParse(s, NumberStyles.HexNumber, Inv, out v)) return false;
        c = new Color(((v >> 16) & 255) / 255f, ((v >> 8) & 255) / 255f, (v & 255) / 255f, alpha);
        return true;
    }

    void MemberRow(Component c, Member m)
    {
        object v;
        try { v = m.Get(c); } catch { return; }
        GUILayout.BeginHorizontal();
        GUILayout.Space(18);
        GUILayout.Label(new GUIContent(Nice(m.Name), m.Name + " · " + m.Type.Name), _propLabel, GUILayout.Width(_propsRect.width * 0.42f), GUILayout.Height(24));
        var key = c.GetInstanceID() + "." + m.Name;
        if (m.Type == typeof(bool))
        {
            var b = (bool)v;
            var nb = MiniSwitch(b);
            if (nb != b) { var comp = c; var name = m.Name; Later(() => EditProp(comp, name, nb ? "true" : "false")); }
        }
        else if (m.Type.IsEnum)
        {
            if (GUILayout.Button(v.ToString(), _chip, GUILayout.Height(24)))
            {
                var vals = Enum.GetValues(m.Type);
                var idx = Array.IndexOf(vals, v);
                var next = vals.GetValue((idx + (Event.current.shift ? vals.Length - 1 : 1)) % vals.Length).ToString();
                var comp = c; var name = m.Name;
                Later(() => EditProp(comp, name, next));
            }
        }
        else if (m.Type == typeof(Color))
        {
            var col = (Color)v;
            var sw = GUILayoutUtility.GetRect(24, 24, GUILayout.Width(24), GUILayout.Height(24));
            Fill(new Rect(sw.x + 2, sw.y + 3, 20, 18), new Color(col.r, col.g, col.b, 1));
            string shown;
            if (!_edit.TryGetValue(key, out shown)) shown = Hex(col);
            var nv = GUILayout.TextField(shown, _field, GUILayout.Height(24));
            Color parsed;
            if (nv != shown) { _edit[key] = nv; if (FromHex(nv, col.a, out parsed)) EditProp(c, m.Name, C(parsed)); }
        }
        else
        {
            string shown;
            if (!_edit.TryGetValue(key, out shown)) shown = Str(v);
            var nv = GUILayout.TextField(shown, _field, GUILayout.Height(24));
            if (nv != shown)
            {
                _edit[key] = nv;
                try { Parse(nv, m.Type); EditProp(c, m.Name, nv.Replace(" ", "")); } catch { }
            }
        }
        GUILayout.EndHorizontal();
    }

    class Holder { public string V; }
    string _lastPropKey;
    float _lastPropAt;
    Cmd _lastPropCmd;
    Holder _lastHolder;

    /// <summary>Изменить свойство с отменой; правки одного поля подряд (набор цифр) — одна отмена.</summary>
    void EditProp(Component c, string member, string value)
    {
        if (c == null) return;
        string old;
        try
        {
            if (member == "enabled") old = EnabledOf(c) == true ? "true" : "false";
            else old = Str(MembersOf(c.GetType()).First(x => x.Name == member).Get(c));
        }
        catch { old = null; }
        SetProp(c, member, value);
        if (old == null || old == value) return;
        var key = c.GetInstanceID() + "." + member;
        var now = Time.realtimeSinceStartup;
        if (key == _lastPropKey && now - _lastPropAt < 1.5f && _undo.Count > 0 && _undo[_undo.Count - 1] == _lastPropCmd)
        {
            _lastHolder.V = value;
            _lastPropAt = now;
            return;
        }
        var holder = new Holder { V = value };
        var target = c;
        Push(Nice(member), () => { if (target != null) SetProp(target, member, old); }, () => { if (target != null) SetProp(target, member, holder.V); });
        _lastHolder = holder;
        _lastPropKey = key;
        _lastPropAt = now;
        _lastPropCmd = _undo[_undo.Count - 1];
    }

    }
