    class Act { public string Id, Group, Name, Icon, Keys, Search; public Action Run; public bool Sel; }

    List<Act> _acts;
    readonly List<string> _recent = new List<string>();

    static readonly string[] ActGroups = { "add", "transform", "arrange", "edit", "color", "look", "behaviour", "physics", "world", "camera", "select", "view", "file", "info" };

    string GroupName(string g) { return T("g." + g, g); }

    void A(string id, string group, string icon, string name, bool sel, Action run, string keys)
    {
        _acts.Add(new Act { Id = id, Group = group, Icon = icon, Name = name, Sel = sel, Run = run, Keys = keys ?? "", Search = (name + " " + GroupName(group) + " " + id).ToLowerInvariant() });
    }

    void A(string id, string group, string icon, string name, bool sel, Action run) { A(id, group, icon, name, sel, run, ""); }

    void RunAct(Act a)
    {
        if (a.Sel && TopSelection().Count == 0) { Status(T("needSel", "Select an object first")); return; }
        try { a.Run(); }
        catch (Exception e) { Status(a.Name + ": " + e.Message); Logger.LogWarning("command " + a.Id + ": " + e); }
        _recent.Remove(a.Id);
        _recent.Insert(0, a.Id);
        if (_recent.Count > 8) _recent.RemoveAt(8);
    }

    Act FindAct(string id) { return Acts().FirstOrDefault(x => x.Id == id); }

    /// <summary>Правка положения выбранного (с отменой).</summary>
    void ForSel(string name, Action<GameObject> f)
    {
        var tops = TopSelection();
        if (tops.Count == 0) { Status(T("needSel", "Select an object first")); return; }
        var before = SnapOf(tops);
        foreach (var g in tops) f(g);
        foreach (var g in tops) Touch(g);
        PushSnap(name, before, SnapOf(tops));
        Dirty();
    }

    class Style0 { public Placed P; public string Mat, Look, Beh, PhysMat; public Color Tint; public bool Tinted, Physics; }

    Style0 StyleOf(Placed p) { return new Style0 { P = p, Mat = p.Mat, Look = p.Look, Beh = BehOf(p), PhysMat = p.PhysMat, Tint = p.Tint, Tinted = p.Tinted, Physics = p.Physics }; }

    static string BehOf(Placed p)
    {
        var b = p.Go != null ? p.Go.GetComponent<ModScriptBehaviour>() : null;
        return b != null ? b.Code() : "";
    }

    void PutStyle(Style0 s)
    {
        var p = s.P;
        if (p.Go == null) return;
        p.Mat = s.Mat; p.Look = s.Look; p.PhysMat = s.PhysMat; p.Tint = s.Tint; p.Tinted = s.Tinted; p.Physics = s.Physics;
        ApplyStyle(p);
        ApplyPhysMat(p.Go, p.PhysMat);
        SetPhysics(p.Go, _simulate && p.Physics);
        SetBeh(p, s.Beh);
    }

    void SetBeh(Placed p, string code)
    {
        var b = p.Go.GetComponent<ModScriptBehaviour>();
        if (b == null && code == "") return;
        if (b == null) b = p.Go.AddComponent<ModScriptBehaviour>();
        b.Paused = false;
        b.Pause();
        b.Parse(code);
        p.Beh = code;
        b.Rebase();
        if (_simulate) b.Paused = false; else { b.Paused = false; b.Pause(); }
    }

    /// <summary>Правка вида поставленных объектов (цвет, материал, поведение, физика) — с отменой.</summary>
    void ForPlaced(string name, Action<Placed> f)
    {
        var ps = TopSelection().Select(PlacedOf).Where(p => p != null).Distinct().ToList();
        if (ps.Count == 0) { Status(TopSelection().Count == 0 ? T("needSel", "Select an object first") : T("onlyYours", "Works on objects you placed")); return; }
        var before = ps.Select(StyleOf).ToList();
        foreach (var p in ps) f(p);
        var after = ps.Select(StyleOf).ToList();
        Push(name, () => { foreach (var s in before) PutStyle(s); }, () => { foreach (var s in after) PutStyle(s); });
        Dirty();
    }

    Bounds AllBounds(List<GameObject> gos)
    {
        var b = BoundsOf(gos[0]);
        foreach (var g in gos) b.Encapsulate(BoundsOf(g));
        return b;
    }

    Vector3 FocusPoint()
    {
        if (_sel != null) return BoundsOf(_sel).center;
        return _cam.transform.position + _cam.transform.forward * 10f;
    }

    /// <summary>Снимок мира (солнце, окружение, туман) для отмены.</summary>
    class WorldSnap
    {
        public World W; public Quaternion SunRot; public float SunI, SunShadow; public Color SunC, Amb, FogC; public float AmbI, FogD; public bool Fog; public UnityEngine.Rendering.AmbientMode Mode;
    }

    WorldSnap SnapWorld()
    {
        var sun = Sun();
        return new WorldSnap
        {
            W = new World { SunIntensity = _world.SunIntensity, SunPitch = _world.SunPitch, SunYaw = _world.SunYaw, Ambient = _world.Ambient, FogDensity = _world.FogDensity, Fog = _world.Fog, SunColor = _world.SunColor, AmbientColor = _world.AmbientColor, FogColor = _world.FogColor, Shadow = _world.Shadow, Weather = _world.Weather, Cycle = _world.Cycle },
            SunRot = sun != null ? sun.transform.rotation : Quaternion.identity, SunI = sun != null ? sun.intensity : 0, SunC = sun != null ? sun.color : Color.white, SunShadow = sun != null ? sun.shadowStrength : 1,
            Amb = RenderSettings.ambientLight, AmbI = RenderSettings.ambientIntensity, Mode = RenderSettings.ambientMode, FogC = RenderSettings.fogColor, FogD = RenderSettings.fogDensity, Fog = RenderSettings.fog,
        };
    }

    void PutWorld(WorldSnap s)
    {
        _world = s.W;
        var sun = Sun();
        if (sun != null) { sun.transform.rotation = s.SunRot; sun.intensity = s.SunI; sun.color = s.SunC; sun.shadowStrength = s.SunShadow; }
        RenderSettings.ambientMode = s.Mode; RenderSettings.ambientLight = s.Amb; RenderSettings.ambientIntensity = s.AmbI;
        RenderSettings.fogColor = s.FogC; RenderSettings.fogDensity = s.FogD; RenderSettings.fog = s.Fog;
        ApplyWeather();
    }

    void ForWorld(string name, Action f)
    {
        var before = SnapWorld();
        f();
        ApplyWorld();
        var after = SnapWorld();
        Push(name, () => PutWorld(before), () => PutWorld(after));
        Dirty();
    }

void TimeOfDay(int h)    {        var sun = Sun();        if (sun == null) { Status(T("world.noSun", "This level has no sun.")); return; }        float pitch, intensity;        Color color;        SunAt(h, out pitch, out intensity, out color);        _world.SunPitch = pitch; _world.SunIntensity = intensity; _world.SunColor = color; _world.SunYaw = sun.transform.eulerAngles.y;        _world.Cycle = -1;    }

    static Color HueColor(int i, int shade)
    {
        var c = Color.HSVToRGB(HueH[i], 0.78f, 0.92f);
        switch (shade)
        {
            case 0: return Color.Lerp(c, Color.white, 0.45f);
            case 2: return c * 0.6f + new Color(0, 0, 0, 0.4f);
            case 3: return c * 0.35f + new Color(0, 0, 0, 0.65f);
            default: return c;
        }
    }

    void EditBeh(Placed p, Action<ModScriptBehaviour> f)
    {
        var b = p.Go.GetComponent<ModScriptBehaviour>();
        if (b == null) { b = p.Go.AddComponent<ModScriptBehaviour>(); b.Rebase(); b.Paused = true; }
        var t = new GameObject("tmp").AddComponent<ModScriptBehaviour>();
        t.Parse(b.Code());
        f(t);
        var code = t.Code();
        Destroy(t.gameObject);
        SetBeh(p, code);
    }

    static string SetSpec(string spec, string key, float v)
    {
        var parts = (spec ?? "").Split(';').Where(x => x != "" && !x.StartsWith(key + "=")).ToList();
        parts.Add(key + "=" + Num(v));
        return string.Join(";", parts.ToArray());
    }

    void RigidProp(string member, string value)
    {
        ForPlaced(T("physics", "Physics"), p => { p.Physics = true; SetPhysics(p.Go, _simulate); });
        foreach (var g in TopSelection())
        {
            var rb = g.GetComponent<Rigidbody>();
            if (rb == null) { rb = g.AddComponent<Rigidbody>(); rb.isKinematic = !_simulate; }
            EditProp(rb, member, value);
        }
    }

    void ColliderProp(string member, string value)
    {
        foreach (var g in TopSelection()) foreach (var c in g.GetComponentsInChildren<Collider>()) EditProp(c, member, value);
    }

    void Kick(Vector3 v)
    {
        if (!_simulate) ToggleSimulate();
        foreach (var g in TopSelection())
        {
            var p = PlacedOf(g);
            if (p != null && !p.Physics) { p.Physics = true; SetPhysics(g, true); }
            var rb = g.GetComponent<Rigidbody>();
            if (rb != null) { rb.isKinematic = false; rb.AddForce(v, ForceMode.VelocityChange); }
        }
    }

    void Explode()
    {
        if (!_simulate) ToggleSimulate();
        var tops = TopSelection();
        var c = AllBounds(tops).center - Vector3.up * 0.5f;
        foreach (var g in tops)
        {
            var p = PlacedOf(g);
            if (p != null && !p.Physics) { p.Physics = true; SetPhysics(g, true); }
            var rb = g.GetComponent<Rigidbody>();
            if (rb != null) { rb.isKinematic = false; rb.AddExplosionForce(900f, c, 20f, 1.5f); }
        }
    }

    void Align(int axis, int where)
    {
        var tops = TopSelection();
        if (tops.Count < 2) { Status(T("needTwo", "Select two objects")); return; }
        var all = AllBounds(tops);
        var target = where == 0 ? all.min[axis] : where == 2 ? all.max[axis] : all.center[axis];
        ForSel(T("move", "Move"), o =>
        {
            var b = BoundsOf(o);
            var cur = where == 0 ? b.min[axis] : where == 2 ? b.max[axis] : b.center[axis];
            var p = o.transform.position; p[axis] += target - cur; o.transform.position = p;
        });
    }

    void Distribute(int axis)
    {
        var tops = TopSelection().OrderBy(g => BoundsOf(g).center[axis]).ToList();
        if (tops.Count < 3) { Status(T("needThree", "Select three or more objects")); return; }
        var a = BoundsOf(tops[0]).center[axis];
        var z = BoundsOf(tops[tops.Count - 1]).center[axis];
        var before = SnapOf(tops);
        for (var i = 1; i < tops.Count - 1; i++)
        {
            var t = a + (z - a) * i / (tops.Count - 1);
            var p = tops[i].transform.position; p[axis] += t - BoundsOf(tops[i]).center[axis]; tops[i].transform.position = p;
            Touch(tops[i]);
        }
        PushSnap(T("move", "Move"), before, SnapOf(tops));
        Dirty();
    }

    void StackOnTop()
    {
        var tops = TopSelection();
        if (tops.Count < 2) { Status(T("needTwo", "Select two objects")); return; }
        var basis = _sel;
        ForSel(T("move", "Move"), o =>
        {
            if (o == basis) return;
            var top = BoundsOf(basis).max.y;
            var b = BoundsOf(o);
            o.transform.position += new Vector3(basis.transform.position.x - o.transform.position.x, top - b.min.y, basis.transform.position.z - o.transform.position.z);
        });
    }

    void AlignToSurface()
    {
        ForSel(T("rotate", "Rotate"), o =>
        {
            var b = BoundsOf(o);
            var cols = o.GetComponentsInChildren<Collider>().Where(c => c.enabled).ToList();
            foreach (var c in cols) c.enabled = false;
            RaycastHit hit;
            var found = Physics.Raycast(new Vector3(b.center.x, b.max.y + 0.5f, b.center.z), Vector3.down, out hit, 2000f, ~0, QueryTriggerInteraction.Ignore);
            foreach (var c in cols) c.enabled = true;
            if (found) o.transform.rotation = Quaternion.FromToRotation(Vector3.up, hit.normal) * Quaternion.Euler(0, o.transform.eulerAngles.y, 0);
        });
    }

    void DuplicateBy(Vector3 offset)

    }

    /// <summary>Копии одного объекта по готовым точкам (круг, сетка, разброс, стопка, спираль) — одной отменой.</summary>
    void CopiesAt(string name, IEnumerable<KeyValuePair<Vector3, Quaternion>> spots)
    {
        if (_sel == null) return;
        var from = _sel;
        var made = new List<Placed>();
        var news = new List<GameObject> { from };
        foreach (var s in spots)
        {
            var c = Clone(from, made);
            if (c == null) continue;
            c.transform.position = s.Key;
            c.transform.rotation = s.Value;
            news.Add(c);
        }
        _selection.Clear();
        _selection.AddRange(news);
        PushMade(name, made);
    }

    void Circle(int n)
    {
        var b = BoundsOf(_sel);
        var r = Mathf.Max(1f, Mathf.Max(b.size.x, b.size.z) * n / (2 * Mathf.PI) * 1.3f);
        var c = _sel.transform.position - Vector3.forward * r;
        var list = new List<KeyValuePair<Vector3, Quaternion>>();
        for (var i = 1; i < n; i++)
        {
            var ang = i * 360f / n;
            var q = Quaternion.Euler(0, ang, 0);
            list.Add(new KeyValuePair<Vector3, Quaternion>(c + q * Vector3.forward * r, q * _sel.transform.rotation));
        }
        CopiesAt(T("arrange", "Arrange"), list);
    }

    void GridArray(int n)
    {
        var b = BoundsOf(_sel);
        var dx = Mathf.Max(0.2f, b.size.x * 1.15f); var dz = Mathf.Max(0.2f, b.size.z * 1.15f);
        var o = _sel.transform.position;
        var list = new List<KeyValuePair<Vector3, Quaternion>>();
        for (var i = 0; i < n; i++) for (var j = 0; j < n; j++) if (i + j > 0) list.Add(new KeyValuePair<Vector3, Quaternion>(o + new Vector3(i * dx, 0, j * dz), _sel.transform.rotation));
        CopiesAt(T("arrange", "Arrange"), list);
    }

    void Scatter(int n, float radius)
    {
        var o = _sel.transform.position;
        var lift = _sel.transform.position.y - BoundsOf(_sel).min.y;
        var list = new List<KeyValuePair<Vector3, Quaternion>>();
        for (var i = 0; i < n; i++)
        {
            var d = UnityEngine.Random.insideUnitCircle * radius;
            var p = o + new Vector3(d.x, 0, d.y);
            RaycastHit hit;
            if (Physics.Raycast(p + Vector3.up * 60f, Vector3.down, out hit, 200f, ~0, QueryTriggerInteraction.Ignore)) p.y = hit.point.y + lift;
            list.Add(new KeyValuePair<Vector3, Quaternion>(p, Quaternion.Euler(0, UnityEngine.Random.Range(0f, 360f), 0) * _sel.transform.rotation));
        }
        CopiesAt(T("arrange", "Arrange"), list);
    }

    void Stack(int n)
    {
        var h = Mathf.Max(0.1f, BoundsOf(_sel).size.y);
        var list = new List<KeyValuePair<Vector3, Quaternion>>();
        for (var i = 1; i < n; i++) list.Add(new KeyValuePair<Vector3, Quaternion>(_sel.transform.position + Vector3.up * h * i, _sel.transform.rotation));
        CopiesAt(T("arrange", "Arrange"), list);
    }

    void Spiral(int n)
    {
        var b = BoundsOf(_sel);
        var r = Mathf.Max(1f, Mathf.Max(b.size.x, b.size.z) * 1.6f);
        var c = _sel.transform.position - Vector3.forward * r;
        var list = new List<KeyValuePair<Vector3, Quaternion>>();
        for (var i = 1; i < n; i++)
        {
            var q = Quaternion.Euler(0, i * 30f, 0);
            list.Add(new KeyValuePair<Vector3, Quaternion>(c + q * Vector3.forward * r + Vector3.up * b.size.y * i * 0.5f, q * _sel.transform.rotation));
        }
        CopiesAt(T("arrange", "Arrange"), list);
    }

    void RestoreLevel()
    {
        var hidden = _touched.Where(t => t.Hidden).ToList();
        if (hidden.Count == 0) { Status(T("nothing", "Nothing found")); return; }
        foreach (var t in hidden) { t.Hidden = false; if (t.Go != null) t.Go.SetActive(true); }
        Push(T("c.restoreLevel", "Bring back"), () => { foreach (var t in hidden) { t.Hidden = true; if (t.Go != null) t.Go.SetActive(false); } }, () => { foreach (var t in hidden) { t.Hidden = false; if (t.Go != null) t.Go.SetActive(true); } });
        Dirty();
    }

    void ViewFrom(Vector3 euler)
    {
        var f = FocusPoint();
        var d = Mathf.Max(6f, Vector3.Distance(_cam.transform.position, f));
        var q = Quaternion.Euler(euler);
        _cam.transform.rotation = q;
        _cam.transform.position = f - q * Vector3.forward * d;
        _yaw = euler.y;
        _pitch = euler.x;
        _orbit = f;
    }

    void GoBookmark(int n)
    {
        KeyValuePair<Vector3, Quaternion> b;

        _cam.transform.rotation = b.Value;
        var e = b.Value.eulerAngles;

        _pitch = e.x > 180 ? e.x - 360 : e.x;
    }

    {
        var tops = TopSelection().Where(g => PlacedOf(g) != null).ToList();
        if (tops.Count == 0) { Status(T("onlyYours", "Works on objects you placed")); return; }
        var center = AllBounds(tops).center;

            {
                Capture(p);
                var pos = p.Parent == "" ? p.Pos - center : p.Pos;
                lines.Add("place|" + p.Id + "|" + Esc(p.Source) + "|" + Esc(p.Name) + "|" + V(pos) + "|" + Q(p.Rot) + "|" + V(p.Scale) + "|" + Esc(p.Mat) + "|" + (p.Tinted ? C(p.Tint) : "") + "|" + (p.Physics ? "phys" : "") + "|" + Esc(p.Beh) + "|" + (tops.Contains(p.Go) ? "" : p.Parent) + "|" + Esc(p.Look) + "|" + Esc(p.PhysMat));
            }

    void InsertSet(string file)
    {
        var aim = Aim();
        var map = new Dictionary<string, Placed>();
        var made = new List<Placed>();
        foreach (var line in File.ReadAllLines(file, Encoding.UTF8))
        {
            var f = line.Split('|');
            if (f.Length < 14 || f[0] != "place") continue;
            var pos = F(f[4]); var rot = F(f[5]); var sc = F(f[6]);
            var p = new Placed { Id = NewId(), Source = f[2], Name = f[3], Mat = f[7], Physics = f[9] == "phys", Beh = f[10], Parent = f[11], Look = f[12], PhysMat = f[13] };
            if (f[8] != "") { var c = F(f[8]); p.Tint = new Color(c[0], c[1], c[2], c[3]); p.Tinted = true; }
            p.Pos = new Vector3(pos[0], pos[1], pos[2]); p.Rot = new Quaternion(rot[0], rot[1], rot[2], rot[3]); p.Scale = new Vector3(sc[0], sc[1], sc[2]);
            p.Go = Make(p.Source, p.Parent == "" ? aim + p.Pos : p.Pos, p.Rot);
            if (p.Go == null) continue;
            map[f[1]] = p;
            Dress(p);
            SetPhysics(p.Go, false);
            var b = p.Go.GetComponent<ModScriptBehaviour>();
            if (b != null) b.Pause();
            _placed.Add(p);
            made.Add(p);
        }
        foreach (var p in made)
        {
            Placed parent;
            if (p.Parent == "" || !map.TryGetValue(p.Parent, out parent)) { p.Parent = ""; continue; }
            p.Parent = parent.Id;
            p.Go.transform.SetParent(parent.Go.transform, false);
            p.Go.transform.localPosition = p.Pos; p.Go.transform.localRotation = p.Rot; p.Go.transform.localScale = p.Scale;
        }
        _selection.Clear();
        foreach (var p in made) if (p.Parent == "") _selection.Add(p.Go);
        PushMade(T("c.insertSetName", "Insert set"), made);
    }

    float _uiUser = 1f;

    string PrefsFile() { return Path.Combine(_root, "studio.cfg"); }

    void LoadPrefs()
    {
        try
        {
            if (!File.Exists(PrefsFile())) return;
            foreach (var line in File.ReadAllLines(PrefsFile()))
            {
                var kv = line.Split('=');
                if (kv.Length != 2) continue;
                // Свой цвет игрока важнее цвета игры; старые настройки с жёлтым по умолчанию цвет игры не перебивают.
                if (kv[0] == "accent") { var c = F(kv[1]); var col = new Color(c[0], c[1], c[2], 1); if (_gameName == "" || Mathf.Abs(col.r - 0.94f) + Mathf.Abs(col.g - 0.71f) + Mathf.Abs(col.b - 0.16f) > 0.03f) { _accentUser = true; SetAccent(col); } }
                if (kv[0] == "ui") _uiUser = Mathf.Clamp(float.Parse(kv[1], Inv), 0.7f, 1.6f);
            }
        }
        catch { }
    }

    void SavePrefs()
    {
        try { File.WriteAllLines(PrefsFile(), _accentUser ? new[] { "accent=" + C(Amber), "ui=" + Num(_uiUser) } : new[] { "ui=" + Num(_uiUser) }); } catch { }
    }

    void SetAccent(Color c)
    {
        Amber = c;
        AmberSoft = new Color(c.r, c.g, c.b, 0.16f);
        _txt = null;
    }
