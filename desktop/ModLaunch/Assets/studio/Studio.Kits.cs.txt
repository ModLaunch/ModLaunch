    readonly Dictionary<string, Material> _colorMats = new Dictionary<string, Material>();

    Material ColorMat(Color c)
    {
        var key = C(c);
        Material m;
        if (_colorMats.TryGetValue(key, out m) && m != null) return m;
        var basis = BaseMaterial();
        if (basis == null) return null;
        m = new Material(basis) { name = "ModScript Studio " + key };
        if (m.HasProperty("_BaseColor")) m.SetColor("_BaseColor", c);
        if (m.HasProperty("_Color")) m.color = c;
        _colorMats[key] = m;
        return m;
    }

    GameObject Prim(Transform root, PrimitiveType type, Vector3 pos, Vector3 scale, Color c, Quaternion rot)
    {
        var go = GameObject.CreatePrimitive(type);
        go.name = type.ToString();
        go.transform.SetParent(root, false);
        go.transform.localPosition = pos;
        go.transform.localRotation = rot;
        go.transform.localScale = scale;
        var m = ColorMat(c);
        if (m != null) go.GetComponent<Renderer>().sharedMaterial = m;
        return go;
    }

    GameObject Box(Transform root, Vector3 center, Vector3 size, Color c) { return Prim(root, PrimitiveType.Cube, center, size, c, Quaternion.identity); }
    GameObject Box(Transform root, Vector3 center, Vector3 size, Color c, Quaternion rot) { return Prim(root, PrimitiveType.Cube, center, size, c, rot); }
    GameObject Cyl(Transform root, Vector3 bottom, float r, float h, Color c) { return Prim(root, PrimitiveType.Cylinder, bottom + Vector3.up * h / 2, new Vector3(r * 2, h / 2, r * 2), c, Quaternion.identity); }
    GameObject Ball(Transform root, Vector3 center, float r, Color c) { return Prim(root, PrimitiveType.Sphere, center, Vector3.one * r * 2, c, Quaternion.identity); }

    GameObject MeshPart(Transform root, Mesh mesh, Vector3 pos, Color c, Quaternion rot)
    {
        var go = new GameObject("part");
        go.transform.SetParent(root, false);
        go.transform.localPosition = pos;
        go.transform.localRotation = rot;
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        var r = go.AddComponent<MeshRenderer>();
        var m = ColorMat(c);
        if (m != null) r.sharedMaterial = m;
        var mc = go.AddComponent<MeshCollider>();
        mc.sharedMesh = mesh;
        mc.convex = true;
        return go;
    }

    /// <summary>Построитель сетки с плоскими гранями: каждый треугольник смотрит наружу от «центра» фигуры.</summary>
    class MeshMaker
    {
        readonly List<Vector3> _v = new List<Vector3>();
        readonly List<int> _t = new List<int>();
        readonly List<Vector2> _uv = new List<Vector2>();
        public Vector3 Center;

        public void Tri(Vector3 a, Vector3 b, Vector3 c)
        {
            var n = Vector3.Cross(b - a, c - a);
            if (Vector3.Dot(n, (a + b + c) / 3f - Center) < 0) { var x = b; b = c; c = x; }
            foreach (var p in new[] { a, b, c }) { _t.Add(_v.Count); _v.Add(p); _uv.Add(new Vector2(p.x + p.z, p.y)); }
        }

        public void Quad(Vector3 a, Vector3 b, Vector3 c, Vector3 d) { Tri(a, b, c); Tri(a, c, d); }

        public Mesh Build()
        {
            var m = new Mesh { name = "ModScript Studio kit" };
            m.vertices = _v.ToArray();
            m.triangles = _t.ToArray();
            m.uv = _uv.ToArray();
            m.RecalculateNormals();
            m.RecalculateBounds();
            return m;
        }
    }

    /// <summary>Выпуклый профиль в плоскости Z–Y, вытянутый по X на ширину w (клин, призма, шестигранник).</summary>
    static Mesh Extrude(Vector2[] profile, float w)
    {
        var mm = new MeshMaker();
        var c = Vector2.zero;
        foreach (var p in profile) c += p;
        c /= profile.Length;
        mm.Center = new Vector3(0, c.y, c.x);
        Func<Vector2, float, Vector3> P = (p, x) => new Vector3(x, p.y, p.x);
        for (var i = 1; i + 1 < profile.Length; i++)
        {
            mm.Tri(P(profile[0], -w / 2), P(profile[i], -w / 2), P(profile[i + 1], -w / 2));
            mm.Tri(P(profile[0], w / 2), P(profile[i], w / 2), P(profile[i + 1], w / 2));
        }
        for (var i = 0; i < profile.Length; i++)
        {
            var a = profile[i]; var b = profile[(i + 1) % profile.Length];
            mm.Quad(P(a, -w / 2), P(b, -w / 2), P(b, w / 2), P(a, w / 2));
        }
        return mm.Build();
    }

    static Mesh Cone(float r, float h, int segs, bool square)
    {
        var mm = new MeshMaker { Center = new Vector3(0, h / 4, 0) };
        var n = square ? 4 : segs;
        var apex = new Vector3(0, h, 0);
        for (var i = 0; i < n; i++)
        {
            var a0 = (i + (square ? 0.5f : 0)) * Mathf.PI * 2 / n;
            var a1 = (i + 1 + (square ? 0.5f : 0)) * Mathf.PI * 2 / n;
            var p0 = new Vector3(Mathf.Cos(a0) * r, 0, Mathf.Sin(a0) * r);
            var p1 = new Vector3(Mathf.Cos(a1) * r, 0, Mathf.Sin(a1) * r);
            mm.Tri(p0, apex, p1);
            mm.Tri(p0, Vector3.zero, p1);
        }
        return mm.Build();
    }

    /// <summary>Прямая призма с n гранями (шестигранная колонна и т. п.).</summary>
    static Mesh Column(float r, float h, int n)
    {
        var mm = new MeshMaker { Center = new Vector3(0, h / 2, 0) };
        for (var i = 0; i < n; i++)
        {
            var a0 = i * Mathf.PI * 2 / n; var a1 = (i + 1) * Mathf.PI * 2 / n;
            var p0 = new Vector3(Mathf.Cos(a0) * r, 0, Mathf.Sin(a0) * r); var p1 = new Vector3(Mathf.Cos(a1) * r, 0, Mathf.Sin(a1) * r);
            mm.Quad(p0, p1, p1 + Vector3.up * h, p0 + Vector3.up * h);
            mm.Tri(Vector3.zero, p0, p1);
            mm.Tri(Vector3.up * h, p0 + Vector3.up * h, p1 + Vector3.up * h);
        }
        return mm.Build();
    }

    static Mesh Dome(float r, int segs)
    {
        var mm = new MeshMaker { Center = new Vector3(0, r * 0.3f, 0) };
        const int rings = 6;
        for (var j = 0; j < rings; j++)
        {
            var t0 = j * Mathf.PI / 2 / rings; var t1 = (j + 1) * Mathf.PI / 2 / rings;
            for (var i = 0; i < segs; i++)
            {
                var a0 = i * Mathf.PI * 2 / segs; var a1 = (i + 1) * Mathf.PI * 2 / segs;
                Func<float, float, Vector3> S = (t, a) => new Vector3(Mathf.Cos(t) * Mathf.Cos(a) * r, Mathf.Sin(t) * r, Mathf.Cos(t) * Mathf.Sin(a) * r);
                if (j == rings - 1) mm.Tri(S(t0, a0), S(t0, a1), new Vector3(0, r, 0));
                else mm.Quad(S(t0, a0), S(t0, a1), S(t1, a1), S(t1, a0));
                if (j == 0) mm.Tri(S(0, a0), S(0, a1), Vector3.zero);
            }
        }
        return mm.Build();
    }

    static Mesh Torus(float R, float r, int segs, int sides)
    {
        var v = new List<Vector3>();
        var t = new List<int>();
        for (var i = 0; i < segs; i++)
            for (var j = 0; j < sides; j++)
            {
                Func<int, int, Vector3> P = (a, b) =>
                {
                    var u = a * Mathf.PI * 2 / segs; var w = b * Mathf.PI * 2 / sides;
                    return new Vector3((R + r * Mathf.Cos(w)) * Mathf.Cos(u), r * Mathf.Sin(w), (R + r * Mathf.Cos(w)) * Mathf.Sin(u));
                };
                var p00 = P(i, j); var p10 = P(i + 1, j); var p11 = P(i + 1, j + 1); var p01 = P(i, j + 1);
                var c = new Vector3(Mathf.Cos((i + 0.5f) * Mathf.PI * 2 / segs) * R, 0, Mathf.Sin((i + 0.5f) * Mathf.PI * 2 / segs) * R);
                foreach (var tri in new[] { new[] { p00, p10, p11 }, new[] { p00, p11, p01 } })
                {
                    var a = tri[0]; var b = tri[1]; var d = tri[2];
                    if (Vector3.Dot(Vector3.Cross(b - a, d - a), (a + b + d) / 3f - c) < 0) { var x = b; b = d; d = x; }
                    t.Add(v.Count); v.Add(a); t.Add(v.Count); v.Add(b); t.Add(v.Count); v.Add(d);
                }
            }
        var m = new Mesh { name = "ModScript Studio torus" };
        m.vertices = v.ToArray();
        m.triangles = t.ToArray();
        m.RecalculateNormals();
        m.RecalculateBounds();
        return m;
    }

    /// <summary>Камень: многогранник со смещёнными вершинами (форма зависит от seed, а значит одинакова при каждой загрузке).</summary>
    static Mesh RockMesh(float s, int seed)
    {
        var mm = new MeshMaker { Center = new Vector3(0, s * 0.35f, 0) };
        const int lat = 5, lon = 8;
        var rnd = new System.Random(seed * 7919 + 13);
        var pts = new Vector3[lat + 1, lon];
        for (var j = 0; j <= lat; j++)
            for (var i = 0; i < lon; i++)
            {
                var t = -Mathf.PI / 2 + j * Mathf.PI / lat;
                var a = i * Mathf.PI * 2 / lon;
                var k = 0.75f + (float)rnd.NextDouble() * 0.45f;
                var y = (Mathf.Sin(t) * 0.5f + 0.5f) * 0.7f * s;
                pts[j, i] = new Vector3(Mathf.Cos(t) * Mathf.Cos(a) * s * 0.5f * k, y, Mathf.Cos(t) * Mathf.Sin(a) * s * 0.5f * k);
            }
        for (var j = 0; j < lat; j++)
            for (var i = 0; i < lon; i++)
                mm.Quad(pts[j, i], pts[j, (i + 1) % lon], pts[j + 1, (i + 1) % lon], pts[j + 1, i]);
        return mm.Build();
    }

    void Glow(GameObject part, Color c)
    {
        foreach (var r in part.GetComponentsInChildren<Renderer>())
            foreach (var m in r.materials)
                if (m.HasProperty("_EmissionColor")) { m.EnableKeyword("_EMISSION"); m.SetColor("_EmissionColor", c * 2.5f); }
    }

    static Light AddLight(Transform root, Vector3 pos, Color c, float range, float intensity, LightType type)
    {
        var go = new GameObject("Light");
        go.transform.SetParent(root, false);
        go.transform.localPosition = pos;
        var l = go.AddComponent<Light>();
        l.type = type;
        l.color = c;
        l.range = range;
        l.intensity = intensity;
        if (type == LightType.Spot) { l.spotAngle = 55; go.transform.localRotation = Quaternion.Euler(90, 0, 0); }
        return l;
    }

    static Color Tone(string name)
    {
        switch (name)
        {
            case "warm": return new Color(1f, 0.78f, 0.5f);
            case "cold": return new Color(0.65f, 0.8f, 1f);
            case "red": return new Color(1f, 0.25f, 0.2f);
            case "green": return new Color(0.3f, 1f, 0.4f);
            case "blue": return new Color(0.3f, 0.5f, 1f);
            case "purple": return new Color(0.7f, 0.35f, 1f);
            case "pink": return new Color(1f, 0.4f, 0.75f);
            case "yellow": return new Color(1f, 0.85f, 0.2f);
            default: return Color.white;
        }
    }

    static ModScriptBehaviour Beh(GameObject go, string code)
    {
        var b = go.GetComponent<ModScriptBehaviour>();
        if (b == null) b = go.AddComponent<ModScriptBehaviour>();
        b.Parse(code);
        b.Rebase();
        return b;
    }
